/** WHO (2025) Implementation Handbook: the binary referral cutoff is 6/12 (LogMAR >= 0.30). */
export const REFERRAL_LOGMAR_CUTOFF = 0.30;

/** WHO (2025) code 88 sentinel: untestable child = mandatory referral. */
export const UNTTESTABLE_LOGMAR = 88;
export const UNTESTABLE_ACUITY_LABEL = '88 (غير قابل للفحص - إحالة وجوبية)';

/** Urgent (48 h) acuity threshold: 6/36 => LogMAR >= 0.80. */
export const URGENT_LOGMAR_CUTOFF = 0.80;

/** Inter-ocular difference threshold that triggers a standard referral. */
export const IOD_LOGMAR_THRESHOLD = 0.20;

/** Asthenopia symptom count that triggers a standard referral (WHO 2025). */
export const ASTHENOPIA_REVIEW_THRESHOLD = 3;

/** External-eye red flags that mandate an urgent (48 h) ophthalmology review. */
export const URGENT_EXTERNAL_EYE_FLAGS = [
  'LEUKOCORIA',
  'CORNEAL_HAZINESS',
  'ANISOCORIA',
  'EYE_MISALIGNMENT',
] as const;

/** External-eye findings that mandate a standard (2-week PHCC) referral. */
export const REVIEW_EXTERNAL_EYE_FLAGS = ['SCLERAL_REDNESS'] as const;

export const LOGMAR_MAP: Record<string, number> = {
  // Censored result at the chart floor. 1.00 is a lower bound, not an exact acuity.
  'أقل من 6/60': 1.00,
  '6/60': 1.00,
  '6/36': 0.80,
  '6/24': 0.60,
  '6/18': 0.50,
  '6/12': 0.30,
  '6/9.5': 0.20,
  '6/7.5': 0.10,
  '6/6': 0.00,
  // WHO (2025) "Untestable Child" code. The child who cannot be tested is a
  // mandatory referral; the value is a sentinel code, not a measured acuity.
  [UNTESTABLE_ACUITY_LABEL]: UNTTESTABLE_LOGMAR,
};

function getObservedLogmar(snellen: string | undefined, recordedLogmar: number | undefined): number | null {
  const label = snellen?.trim();
  if (!label) return null;
  // Tolerate the WHO untestable code written with or without its full label.
  if (label === '88' || label.startsWith('88 ')) return UNTTESTABLE_LOGMAR;
  const mapped = LOGMAR_MAP[label] ?? (label === '6/9' ? 0.2 : undefined);
  if (mapped !== undefined) return mapped;
  return Number.isFinite(recordedLogmar) ? Number(recordedLogmar) : null;
}

/**
 * Worse-seeing eye = Math.max(OD, OS) over valid numeric LogMAR scores.
 * The worse eye is the primary unit of analysis (Murdoch et al.). Returns NaN
 * when neither eye has a valid numeric score (exports as SPSS system-missing).
 */
export function worseEyeLogMAR(
  odLogmar: number | undefined,
  osLogmar: number | undefined
): number {
  const valid = [odLogmar, osLogmar].filter(
    (value): value is number => typeof value === 'number' && Number.isFinite(value)
  );
  if (valid.length === 0) return Number.NaN;
  return Math.max(...valid);
}

/** True when either eye of the pair recorded the WHO untestable code (88). */
export function isUntestablePair(
  odLogmar: number | undefined,
  osLogmar: number | undefined
): boolean {
  return odLogmar === UNTTESTABLE_LOGMAR || osLogmar === UNTTESTABLE_LOGMAR;
}

function binaryFlag(value: unknown): 0 | 1 | null {
  if (value === 0 || value === '0') return 0;
  if (value === 1 || value === '1') return 1;
  return null;
}

export function hasDocumentedReference(record: StudentScreeningRecord): boolean {
  const validDate = typeof record.Gold_Standard_Date === 'string' && !Number.isNaN(Date.parse(record.Gold_Standard_Date));
  return Boolean(
    record.Gold_Standard_Method?.trim() &&
    record.Gold_Standard_Examiner_ID?.trim() &&
    validDate &&
    record.Gold_Standard_Criteria?.trim() &&
    ['YES', 'NO', 'UNKNOWN'].includes(record.Gold_Standard_Masked ?? '')
  );
}

/** Wilson score interval for a binomial proportion, returned as percentages. */
function wilsonInterval95(successes: number, total: number): [number, number] | null {
  if (total <= 0) return null;
  const z = 1.959963984540054;
  const p = successes / total;
  const denominator = 1 + (z * z) / total;
  const center = (p + (z * z) / (2 * total)) / denominator;
  const margin = (z * Math.sqrt((p * (1 - p) / total) + (z * z) / (4 * total * total))) / denominator;
  return [+(Math.max(0, center - margin) * 100).toFixed(1), +(Math.min(1, center + margin) * 100).toFixed(1)];
}

/**
 * Hirschberg Ratio:
 * Approximate conversion used by this model: 1 mm corresponds to about 15 prism diopters.
 * The ratio varies and should not be treated as a definitive clinical measurement.
 */
export function convertHirschbergToPrism(decentrationMM: number): number {
  return Math.round(decentrationMM * 15);
}

/**
 * Calculate physical optotype height in pixels given pxPerMM
 */
export function getOptotypePixelSize(step: AcuityStep, pxPerMM: number, testingDistanceMeters = 2.0): number {
  const distanceFactor = (testingDistanceMeters || 2.0) / 2.0;
  return Math.max(9, Math.round(step.mm * distanceFactor * pxPerMM));
}

/** Normalizes a Snellen / WHO-88 label to its LogMAR value (undefined when unknown). */
export function snellenToLogmar(label: string | undefined | null): number | undefined {
  const value = label?.trim();
  if (!value) return undefined;
  if (value === '88' || value.startsWith('88 ')) return UNTTESTABLE_LOGMAR;
  return LOGMAR_MAP[value] ?? (value === '6/9' ? 0.2 : undefined);
}

/** Extra WHO 2025 / AAPOS inputs that complete the triage verdict. */
export interface ClinicalDecisionContext {
  /** Paper-station (6 m) acuity for the right eye. */
  paperOd?: string;
  /** Paper-station (6 m) acuity for the left eye. */
  paperOs?: string;
  /** Step 3 external-eye red flags (WHO/AAPOS ids). */
  externalEyeFindings?: string[];
  /** WHO 2025 pre-screen answers (history flags, never diagnoses). */
  whoPreScreen?: {
    diabetes?: string;
    eyePainOrSevereItch?: string;
    childAssent?: string;
  } | null;
}

/**
 * Apply the application's configured screening flags. These labels are not clinical
 * diagnoses, but they encode the WHO 2025 / AAPOS referral tiers used by the study:
 *
 * - Urgent (`MODEL_PRIORITY`, 48 h ophthalmology referral): Hirschberg >= 1.0 mm,
 *   MRD1 < 2.5 mm, acuity <= 6/36, WHO code 88 in either station, an external-eye
 *   red flag (leukocoria / corneal haze / anisocoria / misalignment) or severe
 *   eye pain or itch.
 * - Standard (`MODEL_REVIEW`, 2-week PHCC referral): acuity <= 6/12, inter-ocular
 *   difference >= 0.20 LogMAR, scleral redness, known diabetes, or >= 3 asthenopia
 *   symptoms.
 *
 * `context` carries the paper-station acuity, the Step 3 external-eye findings and
 * the WHO pre-screen answers so App.tsx (record commit) and the Step 4 CDSS panel
 * always render the exact same verdict.
 */
export function classifyClinicalDecision(
  odSnellen: string,
  osSnellen: string,
  hirschbergMM: number,
  mrd1MM: number,
  asthenopiaScore: number,
  context: ClinicalDecisionContext = {},
): {
  urgency: 'MODEL_NO_FLAG' | 'MODEL_REVIEW' | 'MODEL_PRIORITY';
  categoryAr: string;
  categoryEn: string;
  detailsAr: string;
  referralCode: number;
} {
  const odLog = snellenToLogmar(odSnellen) ?? 0;
  const osLog = snellenToLogmar(osSnellen) ?? 0;
  const paperOdLog = snellenToLogmar(context.paperOd) ?? 0;
  const paperOsLog = snellenToLogmar(context.paperOs) ?? 0;

  // The worse result across both stations drives the acuity tier.
  const worstLog = Math.max(odLog, osLog, paperOdLog, paperOsLog);
  const smartIod = Math.abs(odLog - osLog);
  const paperIod = context.paperOd && context.paperOs
    ? Math.abs(paperOdLog - paperOsLog)
    : 0;
  const iod = Math.max(smartIod, paperIod);

  const hasStrabismus = hirschbergMM >= 1.0;
  const hasPtosis = mrd1MM < 2.5;

  const findings = context.externalEyeFindings ?? [];
  const urgentFlags = findings.filter((flag) =>
    (URGENT_EXTERNAL_EYE_FLAGS as readonly string[]).includes(flag));
  const reviewFlags = findings.filter((flag) =>
    (REVIEW_EXTERNAL_EYE_FLAGS as readonly string[]).includes(flag));

  const eyePain = context.whoPreScreen?.eyePainOrSevereItch === 'YES';
  const diabetes = context.whoPreScreen?.diabetes === 'YES';

  const untestable = [odSnellen, osSnellen, context.paperOd, context.paperOs]
    .some((label) => snellenToLogmar(label) === UNTTESTABLE_LOGMAR);

  const urgentReasons: string[] = [];
  if (untestable) urgentReasons.push('رمز 88 (غير قابل للفحص الميداني) في إحدى المحطات');
  if (hasStrabismus) urgentReasons.push('عدم تماثل انعكاس قرني ≥ 1.0 مم');
  if (hasPtosis) urgentReasons.push('MRD₁ أقل من 2.5 مم');
  // The WHO-88 sentinel (88 LogMAR) is not a real acuity value — never print it
  // as a numeric LogMAR; the code-88 reason above already states the finding.
  if (!untestable && worstLog >= URGENT_LOGMAR_CUTOFF) {
    urgentReasons.push(`أسوأ حدة بصر ${worstLog.toFixed(2)} LogMAR (≤ 6/36)`);
  }
  if (urgentFlags.length > 0) urgentReasons.push(`علامات فحص خارجي: ${urgentFlags.join('، ')}`);
  if (eyePain) urgentReasons.push('ألم أو حكة شديدة في العين');

  if (urgentReasons.length > 0) {
    return {
      urgency: 'MODEL_PRIORITY',
      categoryAr: 'إحالة عاجلة — مراجعة طبيب عيون خلال 48 ساعة',
      categoryEn: 'Urgent referral - ophthalmology review within 48 hours',
      detailsAr: `أسباب الإحالة العاجلة: ${urgentReasons.join('؛ ')}. وفق WHO 2025 / AAPOS يُحال الطفل إلى طبيب عيون خلال 48 ساعة.`,
      referralCode: 1,
    };
  }

  const reviewReasons: string[] = [];
  if (!untestable && worstLog >= REFERRAL_LOGMAR_CUTOFF) reviewReasons.push(`أسوأ حدة بصر ${worstLog.toFixed(2)} LogMAR (≤ 6/12)`);
  if (iod >= IOD_LOGMAR_THRESHOLD) reviewReasons.push(`فرق العينين ${iod.toFixed(2)} LogMAR`);
  if (reviewFlags.length > 0) reviewReasons.push(`علامات فحص خارجي: ${reviewFlags.join('، ')}`);
  if (diabetes) reviewReasons.push('سكري معروف لدى الطفل');
  if (asthenopiaScore >= ASTHENOPIA_REVIEW_THRESHOLD) reviewReasons.push(`أعراض إجهاد بصري (${asthenopiaScore}/5)`);

  if (reviewReasons.length > 0) {
    return {
      urgency: 'MODEL_REVIEW',
      categoryAr: 'إحالة — مراجعة المركز الصحي خلال أسبوعين',
      categoryEn: 'Standard referral - PHCC review within two weeks',
      detailsAr: `أسسباب المراجعة: ${reviewReasons.join('؛ ')}. تُحال الحالة إلى مركز الصحة الأولية خلال أسبوعين وفق WHO 2025.`,
      referralCode: 1,
    };
  }

  return {
    urgency: 'MODEL_NO_FLAG',
    categoryAr: 'لم تظهر إشارة إحالة وفق حدود النموذج الحالية',
    categoryEn: 'No referral flag under the current model rules',
    detailsAr: `نتائج السلم المسجلة: OD ${odSnellen} وOS ${osSnellen}. غياب إشارة النموذج لا يستبعد مشكلة بصرية ولا يغني عن التقييم السريري عند وجود أعراض أو مخاوف.`,
    referralCode: 0,
  };
}

/**
 * Calculate Advanced Diagnostic Performance & Agreement Statistics:
 * 1. Cohen's Kappa (κ) between Conventional Paper (6m) and Smart Mobile (2m)
 * 2. Sensitivity (%), Specificity (%), PPV (%), NPV (%), and Accuracy (%) against Gold Standard
 */
export function calculateDiagnosticMetrics(records: StudentScreeningRecord[]): DiagnosticKPIs {
  const totalN = records.length;
  if (totalN === 0) {
    return {
      totalN: 0,
      paperCompleteN: 0,
      smartCompleteN: 0,
      pairedN: 0,
      kappa: null,
      kappaInterpretation: 'لا توجد بيانات كافية',
      observedAgreement: 0,
      expectedAgreement: 0,
      contingencyPaperSmart: { bothRefer: 0, paperOnly: 0, smartOnly: 0, bothNormal: 0 },
      goldEvaluatedN: 0,
      sensitivity: null,
      specificity: null,
      ppv: null,
      npv: null,
      accuracy: null,
      sensitivityCI95: null,
      specificityCI95: null,
      ppvCI95: null,
      npvCI95: null,
      verifiedRate: null,
      contingencySmartGold: { tp: 0, fp: 0, fn: 0, tn: 0 },
      timedN: 0,
      meanTimeSavedSec: null,
    };
  }

  // 1. Contingency Paper vs Smart
  let a = 0; // Paper=1, Smart=1
  let b = 0; // Paper=1, Smart=0
  let c = 0; // Paper=0, Smart=1
  let d = 0; // Paper=0, Smart=0
  let totalSavedSec = 0;
  let pairedN = 0;
  let paperCompleteN = 0;
  let smartCompleteN = 0;
  let timedN = 0;

  records.forEach((r) => {
    const paperOd = getObservedLogmar(r.Paper_Snellen_OD, r.Paper_LogMAR_OD);
    const paperOs = getObservedLogmar(r.Paper_Snellen_OS, r.Paper_LogMAR_OS);
    const smartOd = getObservedLogmar(r.Smart_Snellen_OD, r.Smart_LogMAR_OD);
    const smartOs = getObservedLogmar(r.Smart_Snellen_OS, r.Smart_LogMAR_OS);
    const paperComplete = paperOd !== null && paperOs !== null;
    const smartComplete = smartOd !== null && smartOs !== null;
    if (paperComplete) paperCompleteN++;
    if (smartComplete) smartCompleteN++;
    if (!paperComplete || !smartComplete) return;

    // Compare visual-acuity referral outcomes only; ocular triage must not contaminate Smart method agreement.
    // WHO (2025) cutoff: 6/12 => LogMAR >= 0.30; code 88 (untestable) always refers (88 >= 0.30).
    const pRef = Math.max(paperOd, paperOs) >= REFERRAL_LOGMAR_CUTOFF ? 1 : 0;
    const sRef = Math.max(smartOd, smartOs) >= REFERRAL_LOGMAR_CUTOFF ? 1 : 0;
    if (pRef === 1 && sRef === 1) a++;
    else if (pRef === 1 && sRef === 0) b++;
    else if (pRef === 0 && sRef === 1) c++;
    else d++;

    pairedN++;
    if (Number.isFinite(r.Paper_Time_Sec) && Number.isFinite(r.Smart_Time_Sec) && r.Paper_Time_Sec > 0 && r.Smart_Time_Sec > 0) {
      totalSavedSec += r.Paper_Time_Sec - r.Smart_Time_Sec;
      timedN++;
    }
  });

  const observedAgreement = pairedN > 0 ? (a + d) / pairedN : 0;
  const pPaper1 = pairedN > 0 ? (a + b) / pairedN : 0;
  const pPaper0 = pairedN > 0 ? (c + d) / pairedN : 0;
  const pSmart1 = pairedN > 0 ? (a + c) / pairedN : 0;
  const pSmart0 = pairedN > 0 ? (b + d) / pairedN : 0;
  const expectedAgreement = (pPaper1 * pSmart1) + (pPaper0 * pSmart0);

  const kappa = pairedN > 0 && expectedAgreement < 1
    ? (observedAgreement - expectedAgreement) / (1 - expectedAgreement)
    : null;

  let kappaInterpretation = pairedN === 0
    ? 'لا توجد أزواج فحص مكتملة'
    : kappa === null
      ? 'غير محدد: لا يوجد تباين كافٍ'
    : 'اتساق مرتفع (تفسير استرشادي)';
  if (kappa !== null && kappa < 0.20) kappaInterpretation = 'اتساق طفيف (Slight Agreement)';
  else if (kappa !== null && kappa < 0.40) kappaInterpretation = 'اتساق مقبول (Fair Agreement)';
  else if (kappa !== null && kappa < 0.60) kappaInterpretation = 'اتساق متوسط (Moderate Agreement)';
  else if (kappa !== null && kappa < 0.80) kappaInterpretation = 'اتساق جوهري عالي (Substantial Agreement)';

  // 2. Gold Standard Validation Metrics
  let tp = 0; // Smart=1, Gold=1
  let fp = 0; // Smart=1, Gold=0
  let fn = 0; // Smart=0, Gold=1
  let tn = 0; // Smart=0, Gold=0

  const goldRecords = records.filter((r) =>
    binaryFlag(r.Gold_Standard_Confirmed) !== null && hasDocumentedReference(r) &&
    getObservedLogmar(r.Smart_Snellen_OD, r.Smart_LogMAR_OD) !== null &&
    getObservedLogmar(r.Smart_Snellen_OS, r.Smart_LogMAR_OS) !== null
  );
  const goldEvaluatedN = goldRecords.length;

  goldRecords.forEach((r) => {
    // The diagnostic-accuracy endpoint is digital visual acuity alone. The composite
    // model referral flag also includes ocular and symptom inputs.
    const smartOd = getObservedLogmar(r.Smart_Snellen_OD, r.Smart_LogMAR_OD)!;
    const smartOs = getObservedLogmar(r.Smart_Snellen_OS, r.Smart_LogMAR_OS)!;
    // WHO (2025) cutoff 6/12 (LogMAR >= 0.30); code 88 (untestable) always refers.
    const sRef: 0 | 1 = Math.max(smartOd, smartOs) >= REFERRAL_LOGMAR_CUTOFF ? 1 : 0;
    const gRef = binaryFlag(r.Gold_Standard_Confirmed)!;
    if (sRef === 1 && gRef === 1) tp++;
    else if (sRef === 1 && gRef === 0) fp++;
    else if (sRef === 0 && gRef === 1) fn++;
    else tn++;
  });

  const sensitivity = tp + fn > 0 ? +((tp / (tp + fn)) * 100).toFixed(1) : null;
  const specificity = tn + fp > 0 ? +((tn / (tn + fp)) * 100).toFixed(1) : null;
  const ppv = tp + fp > 0 ? +((tp / (tp + fp)) * 100).toFixed(1) : null;
  const npv = tn + fn > 0 ? +((tn / (tn + fn)) * 100).toFixed(1) : null;
  const accuracy = goldEvaluatedN > 0 ? +(((tp + tn) / goldEvaluatedN) * 100).toFixed(1) : null;

  return {
    totalN,
    paperCompleteN,
    smartCompleteN,
    pairedN,
    kappa: kappa === null ? null : +kappa.toFixed(3),
    kappaInterpretation,
    observedAgreement: +(observedAgreement * 100).toFixed(1),
    expectedAgreement: +(expectedAgreement * 100).toFixed(1),
    contingencyPaperSmart: { bothRefer: a, paperOnly: b, smartOnly: c, bothNormal: d },
    goldEvaluatedN,
    sensitivity,
    specificity,
    ppv,
    npv,
    accuracy,
    sensitivityCI95: wilsonInterval95(tp, tp + fn),
    specificityCI95: wilsonInterval95(tn, tn + fp),
    ppvCI95: wilsonInterval95(tp, tp + fp),
    npvCI95: wilsonInterval95(tn, tn + fn),
    verifiedRate: totalN > 0 ? +((goldEvaluatedN / totalN) * 100).toFixed(1) : null,
    contingencySmartGold: { tp, fp, fn, tn },
    timedN,
    meanTimeSavedSec: timedN > 0 ? +(totalSavedSec / timedN).toFixed(1) : null,
  };
}

/**

 * screening different schools offline can never mint the same code.
 *
 *   BAB-101      (legacy, pre-prefix codes are still parsed and accepted)
 *   BAB-A4F-101  (current format: `BAB-<devicePrefix>-<number>`)

 * offline records of several devices are merged in the cloud.

 * so a hand-written code can never be mis-read from paper onto a screen.

/**
 * Derives a stable 3-character device prefix from the device identifier.
 *
 * The prefix is a pure function of `albasar_device_id`, so it never changes
 * for a given handset and is identical across reinstalls of the same device.
 */
export function deviceCodePrefix(deviceId: string): string {
  const source = (deviceId || '').trim().toUpperCase();
  if (!source) return 'XXX';

  // FNV-1a 32-bit — deterministic, dependency-free, and stable across runs.
  let hash = 0x811c9dc5;
  for (let i = 0; i < source.length; i += 1) {
    hash ^= source.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193) >>> 0;
  }

  let prefix = '';
  let value = hash;
  for (let i = 0; i < 3; i += 1) {
    prefix += PREFIX_ALPHABET[value % PREFIX_ALPHABET.length];
    value = Math.floor(value / PREFIX_ALPHABET.length);
  }
  return prefix;
}

 * odd codes (BAB-101, BAB-103, ...) start with the paper 6m chart;
 * even codes (BAB-102, BAB-104, ...) start with the smart 2m test.
 */
export function crossoverSequenceForParticipantNumber(number: number): string {
  return number % 2 === 1 ? PAPER_FIRST_SEQUENCE : DIGITAL_FIRST_SEQUENCE;
}

export function crossoverSequenceForParticipantId(value: unknown): string | null {

 *
 * The numeric tail is derived from the highest `BAB-xxx` number already
 * present in `albasar_master_records_v3` (so it survives export/import
 * cycles), and the code is prefixed with the device signature so two teams
 * working offline in different schools can never produce the same
 * `child_id` — the cloud table's primary key.
 *
 * @param existingIds IDs already stored on this device.
 * @param deviceId    The device identifier whose prefix should be used.
 */
export function generateParticipantId(

  deviceId = '',
): string {
  let highest = PARTICIPANT_ID_FIRST - 1;

}

const luminance = (r: number, g: number, b: number): number => (0.299 * r + 0.587 * g + 0.114 * b) / 255;

const buildLuminanceMap = (data: Uint8ClampedArray, width: number, height: number): Float32Array => {
  const map = new Float32Array(width * height);
  for (let i = 0, p = 0; i < data.length; i += 4, p += 1) {

/** Separable box blur used to suppress tiny specular highlights before detection. */
const boxBlur = (source: Float32Array, width: number, height: number, radius: number): Float32Array => {
  if (radius < 1) return source;
  const horizontal = new Float32Array(width * height);
  const output = new Float32Array(width * height);
  const window = radius * 2 + 1;

  for (let y = 0; y < height; y += 1) {
    for (let x = 0; x < width; x += 1) {
      let sum = 0;
      let count = 0;
      for (let k = -radius; k <= radius; k += 1) {
        const xx = x + k;
        if (xx < 0 || xx >= width) continue;
        sum += source[y * width + xx];
        count += 1;
      }
      horizontal[y * width + x] = sum / count;
    }
  }

  for (let y = 0; y < height; y += 1) {
    for (let x = 0; x < width; x += 1) {
      let sum = 0;
      let count = 0;
      for (let k = -radius; k <= radius; k += 1) {
        const yy = y + k;
        if (yy < 0 || yy >= height) continue;
        sum += horizontal[yy * width + x];
        count += 1;
      }
      output[y * width + x] = sum / count;
    }
  }

  return output;
};

/**
 * Variance of the 4-neighbour Laplacian over the eye band.
 * A low value means the frame is blurred (camera shake or defocus).
 */
export const computeLaplacianVariance = (
  map: Float32Array,
  width: number,
  height: number,
): number => {
  const x0 = EYE_ROI.OD.cx - EYE_ROI.OD.rx - 8;
  const x1 = EYE_ROI.OS.cx + EYE_ROI.OS.rx + 8;
  const y0 = EYE_ROI.OD.cy - EYE_ROI.OD.ry - 8;
  const y1 = EYE_ROI.OD.cy + EYE_ROI.OD.ry + 8;

  let sum = 0;
  let sumSq = 0;
  let count = 0;

  for (let y = Math.max(1, y0); y <= Math.min(height - 2, y1); y += 1) {
    for (let x = Math.max(1, x0); x <= Math.min(width - 2, x1); x += 1) {
      const index = y * width + x;
      const lap = 4 * map[index] - map[index - 1] - map[index + 1] - map[index - width] - map[index + width];
      sum += lap;
      sumSq += lap * lap;
      count += 1;
    }
  }

  if (count === 0) return 0;
  const mean = sum / count;
  return Math.max(0, sumSq / count - mean * mean);
};

interface DarkComponent {
  area: number;
  cx: number;
  cy: number;

/**
 * 8-connected component of dark pixels that contains the darkest pixel of the
 * eye ROI. Isolating one component keeps eyelashes, brows and temple shadows
 * from dragging the pupil centroid off the true pupil.
 */
const collectDarkComponent = (
  map: Float32Array,
  width: number,
  height: number,
  roi: { cx: number; cy: number; rx: number; ry: number },
  threshold: number,
  seedX: number,
  seedY: number,
): DarkComponent => {
  const x0 = Math.max(0, roi.cx - roi.rx);
  const x1 = Math.min(width - 1, roi.cx + roi.rx);
  const y0 = Math.max(0, roi.cy - roi.ry);
  const y1 = Math.min(height - 1, roi.cy + roi.ry);
  const localWidth = x1 - x0 + 1;
  const localHeight = y1 - y0 + 1;

  if (
    localWidth <= 0 || localHeight <= 0
    || seedX < x0 || seedX > x1 || seedY < y0 || seedY > y1
    || map[seedY * width + seedX] > threshold
  ) {
    return { area: 0, cx: seedX, cy: seedY };
  }

  const visited = new Uint8Array(localWidth * localHeight);
  const queue = new Int32Array(localWidth * localHeight);
  const start = (seedY - y0) * localWidth + (seedX - x0);
  visited[start] = 1;
  queue[0] = start;

  let head = 0;
  let tail = 1;
  let area = 0;
  let sumX = 0;
  let sumY = 0;
  let sumWeight = 0;

  while (head < tail) {
    const localIndex = queue[head];
    head += 1;
    const x = x0 + (localIndex % localWidth);
    const y = y0 + Math.floor(localIndex / localWidth);
    const luminance = map[y * width + x];
    const weight = Math.max(0.001, threshold - luminance);

    area += 1;
    sumX += x * weight;
    sumY += y * weight;
    sumWeight += weight;

    for (let dy = -1; dy <= 1; dy += 1) {
      for (let dx = -1; dx <= 1; dx += 1) {
        if (dx === 0 && dy === 0) continue;
        const nx = x + dx;
        const ny = y + dy;
        if (nx < x0 || nx > x1 || ny < y0 || ny > y1) continue;
        if (!insideEllipse(nx, ny, roi)) continue;
        const neighborLocal = (ny - y0) * localWidth + (nx - x0);
        if (visited[neighborLocal]) continue;
        if (map[ny * width + nx] > threshold) continue;
        visited[neighborLocal] = 1;
        queue[tail] = neighborLocal;
        tail += 1;
      }
    }
  }

  return area > 0
    ? { area, cx: sumX / sumWeight, cy: sumY / sumWeight }
    : { area: 0, cx: seedX, cy: seedY };
};

/**
 * Horizontal visible iris diameter, averaged over several rows through the
 * pupil. Each limbus edge is the strongest iris→sclera gradient, which is
 * immune to the small specular reflex sitting inside the iris.
 */
const measureHvid = (
  map: Float32Array,
  width: number,
  height: number,
  roi: { cx: number; cy: number; rx: number; ry: number },
  pupil: ImagePoint,
  pupilRadiusPx: number,
  minLum: number,
  maxLum: number,
): number | null => {
  const dynamicRange = maxLum - minLum;
  const minDistance = Math.max(6, pupilRadiusPx * 1.25);
  const maxDistance = Math.min(roi.rx * 1.02, 88);
  const samples: number[] = [];

  // Robust iris level: median of an annulus just outside the pupil edge.
  const irisSamples: number[] = [];
  for (const dy of [-2, 0, 2]) {
    const row = Math.round(pupil.y + dy);
    if (row < 1 || row >= height - 1) continue;
    for (const side of [-1, 1] as const) {
      for (let d = pupilRadiusPx * 1.3; d <= pupilRadiusPx * 1.7; d += 1) {
        const x = Math.round(pupil.x + side * d);
        if (x < 1 || x >= width - 1) continue;
        if (!insideEllipse(x, row, roi)) continue;
        irisSamples.push(map[row * width + x]);
      }
    }
  }
  const irisLevel = median(irisSamples) ?? minLum + dynamicRange * 0.35;
  const scleraLevel = Math.max(irisLevel + 0.05, maxLum);
  const edgeMidpoint = (irisLevel + scleraLevel) / 2;

  const findLimbus = (row: number, step: 1 | -1): number | null => {
    let previous = map[row * width + Math.round(pupil.x + step * minDistance)];
    for (let dist = minDistance + 1; dist <= maxDistance; dist += 1) {
      const x = Math.round(pupil.x + step * dist);
      if (x < 1 || x >= width - 1) break;
      const value = map[row * width + x];
      if (value >= edgeMidpoint && previous < edgeMidpoint) {
        const next1 = map[row * width + x + step];
        const next2 = map[row * width + x + step * 2];
        if (next1 >= edgeMidpoint && next2 >= edgeMidpoint) {
          // Sub-pixel linear interpolation across the blurred limbus step.
          const t = (edgeMidpoint - previous) / Math.max(1e-6, value - previous);
          return x - step + t * step;
        }
      }
      previous = value;
    }
    return null;
  };

  for (const dy of [0, -2, 2, -4, 4]) {
    const row = Math.round(pupil.y + dy);
    if (row < 1 || row >= height - 1) continue;
    const leftEdge = findLimbus(row, -1);
    const rightEdge = findLimbus(row, 1);
    if (leftEdge === null || rightEdge === null) continue;
    const measured = rightEdge - leftEdge;
    if (
      measured >= Math.max(24, pupilRadiusPx * 1.8)
      && measured <= Math.min(170, roi.rx * 2 * 0.98)
    ) {
      samples.push(measured);
    }
  }

  return median(samples);
};

/**
 * Purkinje (corneal light reflex) localization.
 * The search is constrained to the iris disc so eyelid/sclera highlights can
 * never be mistaken for the reflex, and the winning blob is refined with a
 * luminance-weighted centroid for sub-pixel Hirschberg stability.
 */
const findCornealReflex = (
  map: Float32Array,
  data: Uint8ClampedArray,
  width: number,
  height: number,
  roi: { cx: number; cy: number; rx: number; ry: number },
  pupil: ImagePoint,
  pupilRadiusPx: number,
  hvidPx: number | null,
  minLum: number,
  maxLum: number,
): ImagePoint | null => {
  const dynamicRange = maxLum - minLum;
  if (dynamicRange < 0.1) return null;

  const irisRadius = Math.min(
    roi.rx * 0.95,
    Math.max((hvidPx ?? pupilRadiusPx * 2.4) / 2, pupilRadiusPx * 1.05),
  );
  // A 3 mm Hirschberg deviation is ≈ 24 analysis px; capping the search keeps
  // sclera / limbus highlights out of the candidate set even for large pupils.
  const searchRadius = Math.max(6, Math.min(irisRadius * 0.92, 34));
  const searchRadiusSq = searchRadius * searchRadius;
  const minSpecular = minLum + dynamicRange * 0.7;

  let peakScore = -Infinity;
  let peakX = pupil.x;
  let peakY = pupil.y;
  let peakLum = 0;

  for (let y = Math.max(0, Math.floor(pupil.y - searchRadius)); y <= Math.min(height - 1, Math.ceil(pupil.y + searchRadius)); y += 1) {
    for (let x = Math.max(0, Math.floor(pupil.x - searchRadius)); x <= Math.min(width - 1, Math.ceil(pupil.x + searchRadius)); x += 1) {
      const dx = x - pupil.x;
      const dy = y - pupil.y;
      const distanceSq = dx * dx + dy * dy;
      if (distanceSq > searchRadiusSq) continue;

      const luminance = map[y * width + x];
      if (luminance < minSpecular) continue;

      const index = (y * width + x) * 4;
      const whiteness = Math.min(data[index], data[index + 1], data[index + 2]) / 255;
      if (whiteness < 0.72) continue;

      // Prefer the brightest, whitest pixel and only gently penalise distance
      // so a physiologic kappa offset is not pulled back to the pupil centre.
      const score = luminance + whiteness * 0.08 - (distanceSq / searchRadiusSq) * 0.03;
      if (score > peakScore) {
        peakScore = score;
        peakX = x;
        peakY = y;
        peakLum = luminance;
      }
    }
  }

  if (!Number.isFinite(peakScore)) return null;

  const refineThreshold = Math.max(minLum + dynamicRange * 0.6, peakLum * 0.72);
  let sumWeight = 0;
  let sumX = 0;
  let sumY = 0;
  const refineRadius = 4;

  for (let y = Math.max(0, peakY - refineRadius); y <= Math.min(height - 1, peakY + refineRadius); y += 1) {
    for (let x = Math.max(0, peakX - refineRadius); x <= Math.min(width - 1, peakX + refineRadius); x += 1) {
      if (Math.hypot(x - peakX, y - peakY) > refineRadius + 0.5) continue;
      const luminance = map[y * width + x];
      if (luminance < refineThreshold) continue;
      const weight = (luminance - refineThreshold + 0.02) ** 2;
      sumWeight += weight;
      sumX += x * weight;
      sumY += y * weight;
    }
  }

  return sumWeight > 0
    ? { x: sumX / sumWeight, y: sumY / sumWeight }
    : { x: peakX, y: peakY };
};


/* ------------------------------------------------------------------ *
 * Per-eye landmark detection
 * ------------------------------------------------------------------ */

const detectEyeLandmarks = (
  blurred: Float32Array,
  data: Uint8ClampedArray,
  width: number,
  height: number,
  eye: EyeKey,
): EyeLandmarks => {
  const roi = EYE_ROI[eye];

  // --- Luminance extrema and the darkest-pixel seed -----------------------
  let minLum = 1;
  let maxLum = 0;
  let seedX: number = roi.cx;
  let seedY: number = roi.cy;

  for (let y = Math.max(0, roi.cy - roi.ry); y <= Math.min(height - 1, roi.cy + roi.ry); y += 1) {
    for (let x = Math.max(0, roi.cx - roi.rx); x <= Math.min(width - 1, roi.cx + roi.rx); x += 1) {
      if (!insideEllipse(x, y, roi)) continue;
      const value = blurred[y * width + x];
      if (value < minLum) {
        minLum = value;
        seedX = x;
        seedY = y;
      }
      if (value > maxLum) maxLum = value;
    }
  }

  const dynamicRange = maxLum - minLum;

  // --- Pupil: one connected dark component containing the darkest pixel ---
  const darkThreshold = minLum + dynamicRange * 0.2;
  const component = collectDarkComponent(blurred, width, height, roi, darkThreshold, seedX, seedY);
  const pupilDetected = dynamicRange >= 0.06 && component.area >= 14;
  const pupil: ImagePoint | null = pupilDetected
    ? { x: component.cx, y: component.cy }
    : null;
  const pupilRadiusPx = pupilDetected
    ? Math.min(60, Math.max(3, Math.sqrt(component.area / Math.PI)))
    : null;

  // --- HVID first: it bounds the iris disc used for reflex localization ---
  const hvidPx = pupilDetected && pupil && pupilRadiusPx
    ? measureHvid(blurred, width, height, roi, pupil, pupilRadiusPx, minLum, maxLum)
    : null;

  // --- Corneal reflex (Purkinje): brightest specular blob inside the iris --
  const reflex = pupilDetected && pupil && pupilRadiusPx
    ? findCornealReflex(blurred, data, width, height, roi, pupil, pupilRadiusPx, hvidPx, minLum, maxLum)
    : null;

  // --- Specular glare inside the pupil disc -------------------------------
  let pupilPixels = 0;
  let saturatedPixels = 0;
  if (pupil && pupilRadiusPx) {
    const radius = pupilRadiusPx * 1.15;
    const radiusSq = radius * radius;
    for (let y = Math.max(0, Math.floor(pupil.y - radius)); y <= Math.min(height - 1, Math.ceil(pupil.y + radius)); y += 1) {
      for (let x = Math.max(0, Math.floor(pupil.x - radius)); x <= Math.min(width - 1, Math.ceil(pupil.x + radius)); x += 1) {
        const dx = x - pupil.x;
        const dy = y - pupil.y;
        if (dx * dx + dy * dy > radiusSq) continue;
        pupilPixels += 1;
        const index = (y * width + x) * 4;
        if (data[index] > SATURATION_THRESHOLD && data[index + 1] > SATURATION_THRESHOLD && data[index + 2] > SATURATION_THRESHOLD) {
          saturatedPixels += 1;
        }
      }
    }
  }
  const saturatedPupilRatio = pupilPixels > 0 ? saturatedPixels / pupilPixels : 0;

  // --- Upper lid margin for MRD₁ ------------------------------------------
  let upperLid: ImagePoint | null = null;
  if (pupilDetected && pupil && pupilRadiusPx) {
    const start = Math.max(1, Math.round(pupil.y - pupilRadiusPx * 3.4));
    const end = Math.min(height - 2, Math.round(pupil.y - pupilRadiusPx * 0.35));
    const columns = [-2, -1, 0, 1, 2]
      .map((offset) => Math.round(pupil.x) + offset)
      .filter((x) => x >= 1 && x < width - 1);
    let bestGradient = 0;
    let bestY = -1;
    for (let y = start; y <= end; y += 1) {
      let gradient = 0;
      for (const x of columns) {
        gradient += Math.abs(blurred[(y + 1) * width + x] - blurred[(y - 1) * width + x]);
      }
      if (columns.length > 0) gradient /= columns.length;
      if (gradient > bestGradient) {
        bestGradient = gradient;
        bestY = y;
      }
    }
    if (bestY > 0 && bestGradient >= 0.016) {
      upperLid = { x: pupil.x, y: bestY };
    }
  }

  const offsetPx = pupil && reflex ? distance(pupil, reflex) : null;
  const signedOffsetPx = pupil && reflex ? reflex.x - pupil.x : null;

  return {
    eye,
    pupil,
    reflex,
    upperLid,
    pupilRadiusPx,
    offsetPx,
    signedOffsetPx,
    hvidPx,
    hvidSource: hvidPx ? 'detected' : 'standard',
    saturatedPupilRatio,
    pupilConfidence: pupilDetected ? Math.min(1, component.area / 220) : 0,
  };
};

/**
 * Draws the photo into the canonical analysis frame and returns the mapping used.
 *
 * The frame uses a "cover" fit: the photo is scaled until it fully covers the
 * 600×300 analysis frame and the overflow is cropped around the centre. This keeps
 * the eye band filled for both portrait and landscape framing instead of leaving
 * tiny black letterbox bars that would destroy the sharpness and glare metrics.
 */
export const buildAnalysisFrame = (image: OcularFrameSource): {
  canvas: HTMLCanvasElement;
  transform: AnalysisTransform;
} => {
  const canvas = document.createElement('canvas');
  canvas.width = ANALYSIS_WIDTH;
  canvas.height = ANALYSIS_HEIGHT;
  const ctx = canvas.getContext('2d', { willReadFrequently: true });
  if (!ctx) {
    throw new Error('تعذر تجهيز إطار التحليل للصورة.');
  }

  const imgWidth = (image instanceof HTMLImageElement
    ? image.naturalWidth || image.width
    : image.width) || ANALYSIS_WIDTH;
  const imgHeight = (image instanceof HTMLImageElement
    ? image.naturalHeight || image.height
    : image.height) || ANALYSIS_HEIGHT;
  const scale = Math.max(ANALYSIS_WIDTH / imgWidth, ANALYSIS_HEIGHT / imgHeight);
  const drawWidth = imgWidth * scale;
  const drawHeight = imgHeight * scale;
  const offsetX = (ANALYSIS_WIDTH - drawWidth) / 2;
  const offsetY = (ANALYSIS_HEIGHT - drawHeight) / 2;

  // Full-sensor frames are downsampled into the canonical 600×300 analysis
  // frame. High-quality smoothing preserves sub-pixel limbus/reflex edges.
  ctx.imageSmoothingEnabled = true;
  ctx.imageSmoothingQuality = 'high';
  ctx.fillStyle = '#000000';
  ctx.fillRect(0, 0, ANALYSIS_WIDTH, ANALYSIS_HEIGHT);
  ctx.drawImage(image, offsetX, offsetY, drawWidth, drawHeight);

  return {
    canvas,
    transform: { imgWidth, imgHeight, scale, offsetX, offsetY },
  };
};

/**
 * Full client-side analysis of a captured/uploaded eye photo:
 * sharpness gate, glare gate, per-eye landmarks and the HVID scale factor.
 */
export const analyzeOcularFrame = (image: OcularFrameSource): OcularFrameAnalysis => {
  const { canvas, transform } = buildAnalysisFrame(image);
  const ctx = canvas.getContext('2d', { willReadFrequently: true });
  if (!ctx) throw new Error('تعذر قراءة الصورة للتحليل.');
  const imageData = ctx.getImageData(0, 0, ANALYSIS_WIDTH, ANALYSIS_HEIGHT);
  const raw = buildLuminanceMap(imageData.data, ANALYSIS_WIDTH, ANALYSIS_HEIGHT);
  const blurred = boxBlur(raw, ANALYSIS_WIDTH, ANALYSIS_HEIGHT, 2);

  // Sharpness is measured on the *unblurred* luminance map: the 5×5 box blur used
  // for landmark detection destroys the high-frequency detail that actually encodes
  // motion blur, so scoring the blurred map made every classroom photo look soft.
  const sharpness = computeLaplacianVariance(raw, ANALYSIS_WIDTH, ANALYSIS_HEIGHT);
  const contrast = computeLuminanceStdDev(blurred, ANALYSIS_WIDTH, ANALYSIS_HEIGHT);

  const landmarks = {
    OD: detectEyeLandmarks(blurred, imageData.data, ANALYSIS_WIDTH, ANALYSIS_HEIGHT, 'OD'),
    OS: detectEyeLandmarks(blurred, imageData.data, ANALYSIS_WIDTH, ANALYSIS_HEIGHT, 'OS'),
  };

  const glareRatio = Math.max(landmarks.OD.saturatedPupilRatio, landmarks.OS.saturatedPupilRatio);

  // HVID scale: use the measured iris diameter when both eyes agree, otherwise
  // fall back to the canonical framing reference so the workflow never stalls.
  const detected = [landmarks.OD.hvidPx, landmarks.OS.hvidPx].filter((v): v is number => typeof v === 'number');
  let hvidReference: number | null = null;
  if (detected.length === 2) {
    const [a, b] = detected;
    if (Math.abs(a - b) <= Math.max(12, a * 0.25)) hvidReference = (a + b) / 2;
  } else if (detected.length === 1) {
    hvidReference = detected[0];
  }
  const hvidPx = hvidReference ?? FALLBACK_HVID_PX;

  if (hvidReference) {
    landmarks.OD.hvidPx = hvidReference;
    landmarks.OS.hvidPx = hvidReference;
    landmarks.OD.hvidSource = 'detected';
    landmarks.OS.hvidSource = 'detected';
  }

  const isBlurred = sharpness < LAPLACIAN_VARIANCE_MIN || contrast < CONTRAST_STD_MIN;
  const hasGlare = glareRatio > GLARE_SPAN_THRESHOLD;

  return {
    landmarks,
    sharpness,
    contrast,
    isBlurred,
    hasGlare,
    glareRatio,
    mmPerPx: STANDARD_PEDIATRIC_HVID_MM / hvidPx,
    transform,
    burstCount: 1,
  };
};

const medianPoint = (points: ImagePoint[]): ImagePoint | null => {
  if (points.length === 0) return null;
  const x = median(points.map((point) => point.x));
  const y = median(points.map((point) => point.y));
  return x === null || y === null ? null : { x, y };
};

/**
 * Fuses a short full-resolution burst into one robust measurement.
 *
 * Pupil centre, corneal reflex, upper-lid margin, pupil radius and HVID are the
 * per-coordinate median of the individual frames. Because hand tremor shifts
 * each frame by a different sub-pixel amount, the median cancels transient
 * jitter instead of averaging it into the final Hirschberg/MRD₁ geometry.
 * Quality booleans are recomputed from the median metrics so one bad frame
 * cannot silently open or close the gate.
 */
export const mergeOcularAnalyses = (frames: OcularFrameAnalysis[]): OcularFrameAnalysis => {
  if (frames.length === 0) {
    throw new Error('تعذر دمج الإطارات: لا توجد إطارات صالحة.');
  }
  if (frames.length === 1) {
    return { ...frames[0], burstCount: frames[0].burstCount ?? 1 };
  }

  const numericMedian = (values: number[]): number | null => median(values);

  const detectedHvidValues = frames
    .flatMap((frame) => [frame.landmarks.OD.hvidPx, frame.landmarks.OS.hvidPx])
    .filter((value): value is number => value !== null);
  const hvidReference = numericMedian(detectedHvidValues);
  const hvidPx = hvidReference ?? FALLBACK_HVID_PX;

  const mergeEye = (eye: EyeKey): EyeLandmarks => {
    const eyeFrames = frames.map((frame) => frame.landmarks[eye]);
    const pupils = eyeFrames
      .map((landmarks) => landmarks.pupil)
      .filter((point): point is ImagePoint => point !== null);
    const reflexes = eyeFrames
      .map((landmarks) => landmarks.reflex)
      .filter((point): point is ImagePoint => point !== null);
    const lids = eyeFrames
      .map((landmarks) => landmarks.upperLid)
      .filter((point): point is ImagePoint => point !== null);

    const pupil = medianPoint(pupils);
    const reflex = medianPoint(reflexes);
    const upperLid = medianPoint(lids);
    const pupilRadiusPx = numericMedian(eyeFrames
      .map((landmarks) => landmarks.pupilRadiusPx)
      .filter((value): value is number => value !== null));
    const signedOffsetPx = pupil && reflex ? reflex.x - pupil.x : null;

    return {
      eye,
      pupil,
      reflex,
      upperLid,
      pupilRadiusPx,
      offsetPx: pupil && reflex ? distance(pupil, reflex) : null,
      signedOffsetPx,
      hvidPx: hvidReference,
      hvidSource: hvidReference === null ? 'standard' : 'detected',
      saturatedPupilRatio: numericMedian(eyeFrames.map((landmarks) => landmarks.saturatedPupilRatio)) ?? 0,
      pupilConfidence: numericMedian(eyeFrames.map((landmarks) => landmarks.pupilConfidence)) ?? 0,
    };
  };

  const landmarks = { OD: mergeEye('OD'), OS: mergeEye('OS') };
  const sharpness = numericMedian(frames.map((frame) => frame.sharpness)) ?? 0;
  const contrast = numericMedian(frames.map((frame) => frame.contrast)) ?? 0;
  const glareRatio = Math.max(landmarks.OD.saturatedPupilRatio, landmarks.OS.saturatedPupilRatio);

  return {
    landmarks,
    sharpness,
    contrast,
    isBlurred: sharpness < LAPLACIAN_VARIANCE_MIN || contrast < CONTRAST_STD_MIN,
    hasGlare: glareRatio > GLARE_SPAN_THRESHOLD,
    glareRatio,
    mmPerPx: STANDARD_PEDIATRIC_HVID_MM / hvidPx,
    transform: frames[0].transform,
    burstCount: frames.length,
  };
};

/**
 * Net inter-ocular reflex asymmetry.
 *
 * The measurement uses the **signed horizontal** decentration of the corneal reflex
 * (`reflex.x - pupil.x`) rather than the unsigned pupil↔reflex distance. Two reasons:
 *
 * 1. A purely vertical displacement (hypertropia / hypotropia) must not inflate the
 *    horizontal strabismus metric.
 * 2. The sign separates nasal from temporal deviation. In a frontal (un-mirrored)
 *    capture the patient's right eye (OD) sits on the image left, so its nasal
 *    direction is +x, while OS sits on the image right with a nasal direction of −x.
 *    Mirroring OS onto the same nasal-positive axis makes a nasal deviation in OD
 *    and a temporal deviation in OS show up as a genuine 2× asymmetry instead of
 *    cancelling out as two equal unsigned distances.
 *
 * The physiological 0.5 mm angle-kappa offset (the reflex rests slightly nasal of
 * the pupil centre in a normal eye) is then removed from each eye.
 */
export const computeNetHirschberg = (
  landmarks: Record<EyeKey, EyeLandmarks>,
  mmPerPx: number,
): NetHirschbergResult => {
  const signedOdPx = landmarks.OD.signedOffsetPx
    ?? (landmarks.OD.pupil && landmarks.OD.reflex
      ? landmarks.OD.reflex.x - landmarks.OD.pupil.x
      : null);
  const signedOsPx = landmarks.OS.signedOffsetPx
    ?? (landmarks.OS.pupil && landmarks.OS.reflex
      ? landmarks.OS.reflex.x - landmarks.OS.pupil.x
      : null);

  // Nasal-positive axis (see the note above).
  const odNasalMm = signedOdPx === null ? null : signedOdPx * mmPerPx;
  const osNasalMm = signedOsPx === null ? null : -signedOsPx * mmPerPx;

  const odDeviationMm = odNasalMm === null ? null : odNasalMm - ANGLE_KAPPA_ADJUSTMENT_MM;
  const osDeviationMm = osNasalMm === null ? null : osNasalMm - ANGLE_KAPPA_ADJUSTMENT_MM;

  let asym: number | null = null;
  if (odDeviationMm !== null && osDeviationMm !== null) {
    asym = Math.abs(odDeviationMm - osDeviationMm);
  } else if (odDeviationMm !== null) {
    asym = Math.abs(odDeviationMm);
  } else if (osDeviationMm !== null) {
    asym = Math.abs(osDeviationMm);
  }

  const hirschbergAsymMm = asym === null ? 0 : asym;

  return {
    hirschbergAsymMm,
    prismDiopters: Math.round(hirschbergAsymMm * PRISM_DIOPTERS_PER_MM),
    odOffsetMm: odNasalMm,
    osOffsetMm: osNasalMm,
  };
};

/** MRD₁ estimate from the pupil margin to the detected lower lid margin. */
export const computeMRD1FromLandmarks = (
  landmarks: EyeLandmarks,
  mmPerPx: number,
): number | null => {
  if (!landmarks.upperLid || !landmarks.reflex) return null;
  // MRD₁ = corneal light reflex → upper eyelid margin (validated later).
  const gapPx = landmarks.reflex.y - landmarks.upperLid.y;
  return gapPx * mmPerPx;
};

/** Clamp + flag ocular values that leave the admissible pediatric ranges. */
export const validateOcularRanges = (
  hirschbergMM: number,
  mrd1MM: number,
): OcularRangeValidation => {
  const issues: string[] = [];

  let safeHirschberg = hirschbergMM;
  if (!Number.isFinite(safeHirschberg) || safeHirschberg < HIRSCHBERG_LIMITS.min || safeHirschberg > HIRSCHBERG_LIMITS.max) {
    safeHirschberg = Math.min(HIRSCHBERG_LIMITS.max, Math.max(HIRSCHBERG_LIMITS.min, Number.isFinite(hirschbergMM) ? hirschbergMM : 0));
    issues.push(`قيمة Hirschberg خارج النطاق السريري المسموح (${HIRSCHBERG_LIMITS.min}–${HIRSCHBERG_LIMITS.max} مم)؛ عُدّلت تلقائياً إلى ${safeHirschberg.toFixed(2)} مم.`);
  }

  let safeMrd1 = mrd1MM;
  if (!Number.isFinite(safeMrd1) || safeMrd1 < MRD1_LIMITS.min || safeMrd1 > MRD1_LIMITS.max) {
    safeMrd1 = Math.min(MRD1_LIMITS.max, Math.max(MRD1_LIMITS.min, Number.isFinite(mrd1MM) ? mrd1MM : MRD1_LIMITS.min));
    issues.push(`قيمة MRD₁ خارج النطاق السريري المسموح (${MRD1_LIMITS.min}–${MRD1_LIMITS.max} مم)؛ عُدّلت تلقائياً إلى ${safeMrd1.toFixed(1)} مم.`);
  }

  return { hirschbergMM: +safeHirschberg.toFixed(2), mrd1MM: +safeMrd1.toFixed(1), issues };
};

/** Returns the eye whose region contains the analysis-frame point (nearest wins). */
export const eyeAtAnalysisPoint = (point: ImagePoint): EyeKey | null => {
  const score = (eye: EyeKey) => {
    const roi = EYE_ROI[eye];
    const dx = (point.x - roi.cx) / roi.rx;
    const dy = (point.y - roi.cy) / roi.ry;
    return dx * dx + dy * dy;
  };
  const odScore = score('OD');
  const osScore = score('OS');
  if (odScore > 1.6 && osScore > 1.6) return null;
  return odScore <= osScore ? 'OD' : 'OS';
};

/** Normalizes any supported image source into a data URL string. */
async function resolveSource(
  input: string | Blob | HTMLCanvasElement | HTMLImageElement,
): Promise<string> {
  if (typeof input === 'string') return input;
  if (typeof HTMLCanvasElement !== 'undefined' && input instanceof HTMLCanvasElement) {
    // A single high-quality JPEG frame is far cheaper in memory than a PNG frame
    // and keeps the pupil/reflex edges intact before the final resize.
    return input.toDataURL('image/jpeg', 0.92);
  }
  if (typeof HTMLImageElement !== 'undefined' && input instanceof HTMLImageElement) {
    return input.src;
  }
  return blobToDataUrl(input as Blob);
}

function loadImage(dataUrl: string): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = reject;
    img.src = dataUrl;
  });
}

/** Reads a Blob as a data URL (used to preview / persist the compressed photo). */
export async function blobToDataUrl(blob: Blob): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(reader.result as string);
    reader.onerror = reject;
    reader.readAsDataURL(blob);
  });
}

 * Rationale: a diagnostic-clarity JPEG is ~80–150 KB. Storing that as a data URL
 * inside `localStorage` would blow past the 5 MB quota of the records key and break
 * the 100% offline-first guarantee. IndexedDB stores the raw `Blob` (binary, no
 * base64 inflation) keyed by the study `Child_ID`, so the record payload in
 * localStorage stays small while the photo survives a reload and can still be
 * uploaded to Supabase Storage during the next cloud sync.
 */

const DB_NAME = 'albasar_eye_photos';
const DB_VERSION = 1;
const STORE_NAME = 'photos';

interface PhotoRecord {
  childId: string;
  blob: Blob;
  updatedAt: number;
}

let dbPromise: Promise<IDBDatabase | null> | null = null;

/**
 * Opens the photo database and guarantees that the `photos` object store exists.
 *
 * Self-healing: if a database file already exists at the current version but
 * without the expected store (e.g. it was created by an older build or by
 * another tool that opened the name first), the connection is closed and the
 * database is reopened one version higher so the upgrade handler can create
 * the store. Without this, the photo store would be permanently unusable.
 */
const openDatabase = (): Promise<IDBDatabase | null> => {
  if (typeof indexedDB === 'undefined') return Promise.resolve(null);
  if (dbPromise) return dbPromise;

  dbPromise = new Promise<IDBDatabase | null>((resolve) => {
    const tryOpen = (version: number) => {
      let request: IDBOpenDBRequest;
      try {
        request = indexedDB.open(DB_NAME, version);
      } catch {
        resolve(null);
        return;
      }
      request.onupgradeneeded = () => {
        const db = request.result;
        if (!db.objectStoreNames.contains(STORE_NAME)) {
          db.createObjectStore(STORE_NAME, { keyPath: 'childId' });
        }
      };
      request.onsuccess = () => {
        const db = request.result;
        if (!db.objectStoreNames.contains(STORE_NAME)) {
          db.close();
          if (version < DB_VERSION + 8) {
            tryOpen(version + 1);
            return;
          }
          resolve(null);
          return;
        }
        resolve(db);
      };
      request.onerror = () => resolve(null);
      request.onblocked = () => resolve(null);
    };

    tryOpen(DB_VERSION);
  });

  return dbPromise;
};

/** Stores (or replaces) the compressed ocular photo for one child. */
export async function saveEyePhoto(childId: string, blob: Blob): Promise<boolean> {
  const key = childId.trim();
  if (!key || !blob) return false;
  const db = await openDatabase();
  if (!db) return false;

  return new Promise<boolean>((resolve) => {
    try {
      const tx = db.transaction(STORE_NAME, 'readwrite');
      tx.objectStore(STORE_NAME).put({ childId: key, blob, updatedAt: Date.now() } satisfies PhotoRecord);
      tx.oncomplete = () => resolve(true);
      tx.onerror = () => resolve(false);
      tx.onabort = () => resolve(false);
    } catch {
      resolve(false);
    }
  });
}

/** Reads the compressed ocular photo for one child (null when missing). */
export async function getEyePhoto(childId: string): Promise<Blob | null> {
  const key = childId.trim();
  if (!key) return null;
  const db = await openDatabase();
  if (!db) return null;

  return new Promise<Blob | null>((resolve) => {
    try {
      const tx = db.transaction(STORE_NAME, 'readonly');
      const request = tx.objectStore(STORE_NAME).get(key);
      request.onsuccess = () => {
        const value = request.result as PhotoRecord | undefined;
        resolve(value?.blob ?? null);
      };
      request.onerror = () => resolve(null);
    } catch {
      resolve(null);
    }
  });
}

/** Removes the stored photo for one child (used when a record is discarded). */
export async function deleteEyePhoto(childId: string): Promise<void> {
  const key = childId.trim();
  if (!key) return;
  const db = await openDatabase();
  if (!db) return;
  try {
    const tx = db.transaction(STORE_NAME, 'readwrite');
    tx.objectStore(STORE_NAME).delete(key);
  } catch {
    // Non-fatal: the photo simply stays in the store.
  }
}

export function parseRecordBackup(text: string): StudentScreeningRecord[] {
  if (text.length > 25_000_000) throw new Error('حجم النسخة الاحتياطية يتجاوز 25 ميغابايت.');
  let parsed: unknown;
  try {
    parsed = JSON.parse(text);
  } catch {
    throw new Error('الملف ليس JSON صالحاً. استخدم نسخة JSON الاحتياطية الصادرة من النظام.');
  }

  const rows = Array.isArray(parsed)
    ? parsed
    : parsed && typeof parsed === 'object' && Array.isArray((parsed as { records?: unknown }).records)
      ? (parsed as { records: unknown[] }).records

function stableSerialize(value: unknown): string {
  if (Array.isArray(value)) return `[${value.map(stableSerialize).join(',')}]`;
  if (value && typeof value === 'object') {
    const entries = Object.entries(value as Record<string, unknown>)
      .sort(([left], [right]) => left.localeCompare(right));
    return `{${entries.map(([key, child]) => `${JSON.stringify(key)}:${stableSerialize(child)}`).join(',')}}`;
  }
  return JSON.stringify(value) ?? 'null';
}

export function analyzeRecordMerge(
  existing: StudentScreeningRecord[],
  incoming: StudentScreeningRecord[]
): RecordMergeAnalysis {
  const existingById = new Map<string, StudentScreeningRecord>();
  const conflicts = new Set<string>();
  for (const record of existing) {
    const key = normalizeParticipantId(record.Child_ID);
    const prior = existingById.get(key);
    if (prior && stableSerialize(prior) !== stableSerialize(record)) conflicts.add(record.Child_ID.trim());
    else if (!prior) existingById.set(key, record);
  }

  const seenIncoming = new Map<string, StudentScreeningRecord>();
  const additions: StudentScreeningRecord[] = [];
  let exactDuplicates = 0;
  for (const record of incoming) {
    const key = normalizeParticipantId(record.Child_ID);
    const previousImport = seenIncoming.get(key);
    if (previousImport) {
      if (stableSerialize(previousImport) === stableSerialize(record)) exactDuplicates++;
      else conflicts.add(record.Child_ID.trim());
      continue;
    }
    seenIncoming.set(key, record);

    const previousLocal = existingById.get(key);
    if (!previousLocal) additions.push(record);
    else if (stableSerialize(previousLocal) === stableSerialize(record)) exactDuplicates++;
    else conflicts.add(record.Child_ID.trim());
  }

  return { additions, exactDuplicates, conflicts: Array.from(conflicts), importedCount: incoming.length };
}

export function createRecordBackup(records: StudentScreeningRecord[]) {
  return {
    application: 'Al-Basar AI',
    schemaVersion: 3,
    exportedAt: new Date().toISOString(),
    records,
  };
}

const playTone = (frequency: number, duration = 0.08, volume = 0.03) => {
  try {
    const AudioContextClass = window.AudioContext || (window as any).webkitAudioContext;
    if (!AudioContextClass) return;
    const context = new AudioContextClass();
    const oscillator = context.createOscillator();
    const gain = context.createGain();
    oscillator.type = 'sine';
    oscillator.frequency.value = frequency;
    gain.gain.setValueAtTime(0.0001, context.currentTime);
    gain.gain.exponentialRampToValueAtTime(volume, context.currentTime + 0.01);
    gain.gain.exponentialRampToValueAtTime(0.0001, context.currentTime + duration);
    oscillator.connect(gain);
    gain.connect(context.destination);
    oscillator.start();
    oscillator.stop(context.currentTime + duration + 0.02);
    oscillator.onended = () => context.close();
  } catch {}
};

   * concluded immediately with the best line actually passed.

  if (!video.videoWidth || !video.videoHeight) return null;

/**
 * Applies supported `advanced` track constraints. If a browser rejects the
 * combined set, each capability is retried alone so partial AE/AF support
 * still benefits from the lock instead of failing the whole capture flow.
 */
const applyAdvancedTrackConstraints = async (
  track: MediaStreamTrack,
  constraints: ExtendedTrackConstraintSet,
): Promise<string[]> => {
  const keys = Object.keys(constraints).filter((key) => (
    constraints[key as keyof ExtendedTrackConstraintSet] !== undefined
  ));
  if (keys.length === 0) return [];

  try {
    await track.applyConstraints({ advanced: [constraints] } as MediaTrackConstraints);
    return keys;
  } catch {
    const applied: string[] = [];
    for (const key of keys) {
      const value = constraints[key as keyof ExtendedTrackConstraintSet];
      if (value === undefined) continue;
      try {
        await track.applyConstraints({
          advanced: [{ [key]: value }],
        } as MediaTrackConstraints);
        applied.push(key);
      } catch {
        // Capability is advertised but the device rejects this mode.
      }
    }
    return applied;
  }
};

/**
 * Canvas detection can fail in extreme lighting or on a locked-down browser.
 * The examiner must still be able to tap the two visible landmarks manually,
 * so we provide a neutral HVID-scaled frame that the HUD can write into.
 */
const createFallbackAnalysis = (image: HTMLImageElement): OcularFrameAnalysis => {
  const emptyLandmark = (eye: EyeKey): EyeLandmarks => ({
    eye,
    pupil: null,
    reflex: null,
    upperLid: null,
    pupilRadiusPx: null,
    offsetPx: null,
    signedOffsetPx: null,
    hvidPx: null,
    hvidSource: 'standard',
    saturatedPupilRatio: 0,
    pupilConfidence: 0,
  });

  return {
    landmarks: { OD: emptyLandmark('OD'), OS: emptyLandmark('OS') },
    sharpness: 0,
    contrast: 0,
    isBlurred: false,
    hasGlare: false,
    glareRatio: 0,
    mmPerPx: STANDARD_PEDIATRIC_HVID_MM / FALLBACK_HVID_PX,
    transform: {
      imgWidth: image.naturalWidth || ANALYSIS_WIDTH,
      imgHeight: image.naturalHeight || ANALYSIS_HEIGHT,
      scale: 1,
      offsetX: 0,
      offsetY: 0,
    },
  };
};

const toEyeLandmarks = (
  eye: EyeKey,
  points: AuditPointSet,
  base: EyeLandmarks,
): EyeLandmarks => {
  const offsetPx = points.pupil && points.reflex
    ? Math.hypot(points.pupil.x - points.reflex.x, points.pupil.y - points.reflex.y)
    : null;
  const signedOffsetPx = points.pupil && points.reflex
    ? points.reflex.x - points.pupil.x
    : null;

  return {
    ...base,
    eye,
    pupil: points.pupil,
    reflex: points.reflex,
    upperLid: points.eyelid,
    offsetPx,
    signedOffsetPx,
  };
};

      if (capabilities.focusMode?.includes('continuous')) constraints.focusMode = 'continuous';
      if (capabilities.exposureMode?.includes('continuous')) constraints.exposureMode = 'continuous';
    }

    if (Object.keys(constraints).length === 0) {
      cameraLockedRef.current = false;
      setCameraLockLabel('');
      return false;
    }

    if (!locked) {
      cameraLockedRef.current = false;

      return appliedKeys.length > 0;

   * Frame-to-frame luminance differencing on a tiny 80×45 sample. It detects
   * camera shake and exposure jumps without touching the full-resolution
   * capture path, then locks AE/AF when the scene has been stable long enough.
   */
  const startStabilityMonitor = useCallback(() => {
    stopStabilityMonitor();
    const video = videoRef.current;
    if (!video) return;

    const sampleCanvas = document.createElement('canvas');
    sampleCanvas.width = 80;
    sampleCanvas.height = 45;
    stabilitySampleCanvasRef.current = sampleCanvas;
    const tracker = stabilityTrackerRef.current;
    tracker.state = 'unstable';
    tracker.smoothedMotion = 1;
    setStabilityState('unstable');

    function schedule() {
      if (stabilitySampleCanvasRef.current !== sampleCanvas) return;
      stabilityMonitorRef.current = window.setTimeout(tick, STABILITY_SAMPLE_MS);
    }

    function tick() {
      stabilityMonitorRef.current = null;
      const activeVideo = videoRef.current;
      const canvas = stabilitySampleCanvasRef.current;
      if (!activeVideo || !canvas) return;
      if (activeVideo.readyState < 2 || activeVideo.videoWidth === 0) {
        schedule();
        return;
      }

      const context = canvas.getContext('2d', { willReadFrequently: true });
      if (!context) {
        schedule();
        return;
      }

      context.drawImage(activeVideo, 0, 0, canvas.width, canvas.height);
      const pixels = context.getImageData(0, 0, canvas.width, canvas.height).data;
      const luma = new Float32Array(canvas.width * canvas.height);
      let sum = 0;
      for (let index = 0, pixel = 0; index < pixels.length; index += 4, pixel += 1) {
        const value = (0.299 * pixels[index] + 0.587 * pixels[index + 1] + 0.114 * pixels[index + 2]) / 255;
        luma[pixel] = value;
        sum += value;
      }
      const mean = sum / luma.length;
      const now = performance.now();
      const state = stabilityTrackerRef.current;

      if (state.previous && state.previous.length === luma.length) {
        let absoluteDifference = 0;
        for (let index = 0; index < luma.length; index += 1) {
          absoluteDifference += Math.abs(luma[index] - state.previous[index]);
        }
        const frameDelta = absoluteDifference / luma.length;
        const exposureDelta = Math.max(0, Math.abs(mean - state.previousMean) - 0.02);
        const motion = frameDelta + exposureDelta * 0.8;
        state.smoothedMotion = state.smoothedMotion * 0.55 + motion * 0.45;

        const threshold = state.state === 'stable'
          ? STABILITY_UNSTABLE_MOTION
          : STABILITY_STABLE_MOTION;
        const stableNow = state.smoothedMotion < threshold;

        if (stableNow) {
          state.unstableStreak = 0;
          if (state.stableSince === 0) state.stableSince = now;
          const settled = now - state.stableSince >= STABILITY_SETTLE_MS;
          if (state.state !== 'stable' && settled && !burstActiveRef.current) {
            state.state = 'stable';
            setStabilityState('stable');
            if (!state.locked && !state.locking && now >= state.lockSuppressedUntil) {
              state.locking = true;
              void setCameraLock(true).then((applied) => {
                state.locking = false;
                if (applied) {
                  state.locked = true;
                  state.lockedAt = performance.now();
                } else {
                  state.lockSuppressedUntil = performance.now() + CAMERA_LOCK_SUPPRESS_MS;
                }
              });
            }
          }
        } else {
          state.stableSince = 0;
          state.unstableStreak += 1;
          if (
            state.state === 'stable'
            && state.unstableStreak >= STABILITY_UNSTABLE_STREAK
            && now - state.lockedAt >= CAMERA_LOCK_GRACE_MS
          ) {
            state.state = 'unstable';
            setStabilityState('unstable');
            if (state.locked) {
              state.locked = false;
              state.lockSuppressedUntil = now + CAMERA_LOCK_SUPPRESS_MS;
              void setCameraLock(false);
            }
          }
        }
      }

      state.previous = luma;
      state.previousMean = mean;
      schedule();
    }

    stabilityMonitorRef.current = window.setTimeout(tick, 40);
  }, [setCameraLock, stopStabilityMonitor]);

      notify(message, 'error');

    if (videoRef.current?.srcObject) {
      const existing = videoRef.current.srcObject as MediaStream;
      existing.getTracks().forEach((track) => track.stop());
      videoRef.current.srcObject = null;

    cameraTrackRef.current = null;
    cameraCapabilitiesRef.current = null;
    cameraLockedRef.current = false;
    setCameraLockLabel('');

    setCameraStarting(true);
    setCameraError('');

    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: {
          facingMode: { ideal: facing },
          width: { ideal: 1920 },
          height: { ideal: 1080 },
          frameRate: { ideal: 30 },

      });
      const video = videoRef.current;
      if (!video) {
        stream.getTracks().forEach((track) => track.stop());
        const message = 'تعذر تجهيز معاينة الكاميرا. استخدم زر رفع الصورة.';
        setCameraError(message);
        notify(message, 'error');
        return;
      }
      video.srcObject = stream;
      video.setAttribute('playsinline', 'true');
      await video.play();
      await configureCameraTrack(stream, facing);
      setCameraActive(true);
      setCameraFacingMode(facing);
      startStabilityMonitor();
    } catch {
      stopStabilityMonitor();
      const video = videoRef.current;
      if (video?.srcObject) {
        const stream = video.srcObject as MediaStream;
        stream.getTracks().forEach((track) => track.stop());
        video.srcObject = null;

      cameraTrackRef.current = null;
      cameraCapabilitiesRef.current = null;
      cameraLockedRef.current = false;
      setCameraLockLabel('');
      const message = 'تعذر فتح الكاميرا. تحقق من الصلاحيات أو استخدم رفع الصورة.';
      setCameraError(message);
      notify(message, 'error');

      setCameraStarting(false);

   * 3-frame full-resolution burst (0 / 150 / 300 ms). Each frame is analysed
   * independently and the landmark coordinates are spatially median-fused.

      return;

    setIsShutterActive(true);

    setAutoStatus('scanning');
    setAutoMessage(`جارٍ التقاط ${BURST_FRAME_COUNT} إطارات كاملة الدقة ودمجها مكانياً...`);

    try {
      if (!cameraLockedRef.current) {
        const lockedNow = await setCameraLock(true);
        if (lockedNow) await waitFor(180);
      }

      const frames: HTMLCanvasElement[] = [];
      for (let index = 0; index < BURST_FRAME_COUNT; index += 1) {
        if (index > 0) await waitFor(BURST_INTERVAL_MS);
        const frame = captureVideoFrame(video);
        if (frame) frames.push(frame);
      }
      if (frames.length === 0) {
        throw new Error('تعذر التقاط إطار صالح من الكاميرا. أعد المحاولة أو استخدم رفع الصورة.');
      }

      setAutoMessage(`جارٍ تحليل ${frames.length} إطارات واستخراج الحدقة والانعكاس...`);
      await waitFor(0);
      const analyses = frames.map((frame) => analyzeOcularFrame(frame));
      const merged = mergeOcularAnalyses(analyses);
      const sharpestIndex = analyses.reduce(
        (bestIndex, current, index) => (
          current.sharpness > analyses[bestIndex].sharpness ? index : bestIndex
        ),
        0,
      );
      const preview = await canvasToImage(frames[sharpestIndex]);

      stopCamera();
      if (userImage) hadRetakeRef.current = true;
      setUserImage(preview);
      setAnalysis(merged);
      setAuditPoints(pointsFromAnalysis(merged));
      setQualityState('idle');
      setQualityMessage('');
      setAIQualityFlag(0);
      setHirschbergRecorded(false);
      setMrd1Recorded(false);
      lidWarningShownRef.current = false;
      setAutoStatus('scanning');
      setAutoMessage(`تم دمج ${frames.length} إطارات — جارٍ تطبيق بوابة الجودة...`);
      await waitFor(0);
      // The photo is always captured with a single tap: compress the sharpest
      // full-resolution frame straight away, before any quality advisory.
      void emitCompressedPhoto(frames[sharpestIndex]);
      commitAnalysisResult(merged, { burstCount: frames.length });
    } catch (error) {
      const message = error instanceof Error
        ? error.message
        : 'تعذر التقاط الصورة. حاول مرة أخرى أو استخدم رفع الصورة.';
      setAutoStatus('error');
      setAutoMessage(message);
      setQualityState('idle');
      notify(message, 'error');
    } finally {
      setIsShutterActive(false);
      burstActiveRef.current = false;
      setBurstActive(false);
      void setCameraLock(false);
    }

    const file = event.target.files?.[0];
    event.target.value = '';
    if (!file) return;

      image.onload = () => {
        stopCamera();
        if (userImage) hadRetakeRef.current = true;
        processLoadedImage(image);
      };
      image.src = e.target?.result as string;
    };
    reader.readAsDataURL(file);
  };

    setHirschbergRecorded(false);
    setMrd1Recorded(false);

    if (x < 0 || x > ANALYSIS_WIDTH || y < 0 || y > ANALYSIS_HEIGHT) return;

      y: Math.max(0, Math.min(ANALYSIS_HEIGHT, y)),

      [eye]: {

      setAutoStatus('done');

      setAutoMessage(outcome.mrdAvailable

      setQualityMessage('');

    setMrd1MM(preset.mrd1MM);
    setHirschbergRecorded(true);
    setMrd1Recorded(true);

    notify(`تم تطبيق الإعداد السريري: ${preset.label}. يمكنك تعديل القيم يدوياً إذا لزم.`, 'success');
  };

    setExternalEyeGateAbnormal(false);
    setExternalEyeFindings([]);
    setExternalEyeOther('');
    setExternalEyeScreened(true);

  };

  const alignmentState = !hirschbergRecorded || !mrd1Recorded ? 'idle' : hasSuspicion ? 'warn' : 'pass';
  const prismDiopters = convertHirschbergToPrism(hirschbergMM);

  const renderEyeAuditHud = (eye: EyeKey) => {
    const points = auditPoints[eye];
    return (
      <g key={eye}>
        {points.pupil && (
          <g>
            <line
              x1={points.pupil.x - 12}
              y1={points.pupil.y}
              x2={points.pupil.x + 12}
              y2={points.pupil.y}
              stroke="#ef4444"
              strokeWidth="2.5"
              strokeLinecap="round"
              vectorEffect="non-scaling-stroke"
            />
            <line
              x1={points.pupil.x}
              y1={points.pupil.y - 12}
              x2={points.pupil.x}

              stroke="#ef4444"

            <circle

              cy={points.pupil.y}

              strokeWidth="1.5"

            />
            <text x={points.pupil.x + 16} y={points.pupil.y - 12} fill="#fecaca" fontSize="12" fontWeight="800" stroke="#450a0a" strokeWidth="2.5" paintOrder="stroke">
              {eye}

          </g>
        )}
        {points.reflex && (
          <g>
            <circle
              cx={points.reflex.x}
              cy={points.reflex.y}
              r="8"
              fill="rgba(34,211,238,0.16)"
              stroke="#22d3ee"
              strokeWidth="2"
              vectorEffect="non-scaling-stroke"
            />
            <circle cx={points.reflex.x} cy={points.reflex.y} r="2.7" fill="#22d3ee" />
          </g>
        )}
        {points.eyelid && (
          <g>
            <line
              x1={points.eyelid.x - 11}
              y1={points.eyelid.y}

  return (

            إضاءة موزعة
          </span>

        <div className="capture-stage">
          <video

            playsInline

                <svg viewBox="0 0 600 300" preserveAspectRatio="none">
                  <ellipse cx="210" cy="150" rx="98" ry="58" fill="none" stroke="#7dd3fc" strokeWidth="2.5" strokeDasharray="8 5" />
                  <ellipse cx="390" cy="150" rx="98" ry="58" fill="none" stroke="#86efac" strokeWidth="2.5" strokeDasharray="8 5" />
                  <line x1="308" y1="150" x2="292" y2="150" stroke="#f8fafc" strokeWidth="2" />
                  <line x1="300" y1="105" x2="300" y2="195" stroke="rgba(248,250,252,.36)" strokeWidth="1.5" strokeDasharray="5 5" />

              {isShutterActive && <div className="capture-shutter" />}
