-- Al-Basar AI v3 · Supabase schema (SPSS 35+ variable mirror)
-- Idempotent: safe to re-run inside Supabase → SQL Editor.
-- Local storage remains the primary source during offline screening.
--
-- Field sync requires a real, signed-in Supabase session.
-- The `anon` role is no longer granted any privilege on the records table:
-- RLS scopes every row to `owner_id = auth.uid()` (or a supervisor whose
-- JWT carries `app_metadata.role = 'admin'`). A publishable key can never
-- pull the children's dataset. Offline work stays in the device's local
-- store until the examiner signs in.
-- =========================================================

-- ---------------------------------------------------------------------------
-- 🔧 تصحيح نوع عمودَي الزمن (SQLSTATE 22P02)
--
-- `add column if not exists` هو no-op إذا كان العمود موجوداً، لذا لا يكفي
-- تغيير النوع في قائمة الإضافة: يجب تغيير نوع العمود القائم صراحةً.
--
-- السبب: المؤقتان يسجّلان الزمن بعُشر الثانية
-- (`setPaperTimeSec(+elapsed.toFixed(1))` / `setSmartTimeSec(+elapsed.toFixed(1))`)،
-- فتكتب قيمة مثل 74.5 بينما العمود كان `integer` — فيرفضه PostgreSQL بـ
--   400 (22P02): invalid input syntax for type integer: "74.5"
-- والتحويل إلى `numeric` يحفظ دقة الأجزاء العشرية التي يحتاجها تحليل
-- المقارنة الزمنية في SPSS.
--
-- ⚠️ هذا البيان يُعيد كتابة الجدول (ACCESS EXCLUSIVE lock) — على جدول
--   صغير فوري، وعلى جدول كبير نفّذه في نافذة صيانة.
-- ---------------------------------------------------------------------------
alter table public.student_screening_records

-- Keep updated_at accurate for supervisor aggregation.
create or replace function public.set_updated_at()
returns trigger language plpgsql as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists trg_ssr_updated_at on public.student_screening_records;

before update on public.student_screening_records
for each row execute function public.set_updated_at();

-- =========================================================
-- Examiner accounts and roles (optional authenticated features)
-- =========================================================
create table if not exists public.examiner_profiles (
  user_id uuid primary key references auth.users(id) on delete cascade,
  role text not null default 'examiner' check (role in ('examiner', 'supervisor', 'admin')),
  display_name text,
  examiner_code text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

alter table public.examiner_profiles enable row level security;

-- Security-definer helper: avoids recursive RLS lookups on examiner_profiles.
-- The role is accepted from EITHER source:
--   * `app_metadata.role` in the JWT  — the authoritative, server-only source.
--   * the mirrored `examiner_profiles.role` row — kept in sync by the trigger
--     below so the table stays useful for reporting.
create or replace function public.is_supervisor()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select
    coalesce((auth.jwt() -> 'app_metadata' ->> 'role') = 'admin', false)
    or exists (
      select 1
      from public.examiner_profiles
      where user_id = auth.uid()
        and role in ('supervisor', 'admin')
    );
$$;

revoke all on function public.is_supervisor() from public;
grant execute on function public.is_supervisor() to authenticated;

-- Every new Auth user receives an examiner profile automatically.
-- The role is copied from `raw_app_meta_data`, which the user cannot modify.
create or replace function public.handle_new_examiner()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  insert into public.examiner_profiles (user_id, display_name, role)
  values (
    new.id,
    coalesce(new.raw_user_meta_data ->> 'display_name', new.email),
    case when new.raw_app_meta_data ->> 'role' = 'admin' then 'admin' else 'examiner' end
  )
  on conflict (user_id) do nothing;
  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
after insert on auth.users
for each row execute function public.handle_new_examiner();

drop policy if exists "albasar profile self read" on public.examiner_profiles;
create policy "albasar profile self read"
  on public.examiner_profiles
  for select
  to authenticated
  using (user_id = auth.uid() or public.is_supervisor());

grant usage on schema public to authenticated;
grant usage on schema public to anon;
grant select on public.examiner_profiles to authenticated;
revoke all on public.examiner_profiles from anon;

-- =========================================================
-- Server-trusted RBAC helper
-- =========================================================
-- app_metadata lives in auth.users.raw_app_meta_data and is writable ONLY by
-- the Supabase dashboard / a service_role key. `user_metadata` is writable by
-- the user through auth.updateUser(), so it is never consulted for a role.
create or replace function public.is_albasar_admin()
returns boolean
language sql
stable
as $$
  select coalesce((auth.jwt() -> 'app_metadata' ->> 'role') = 'admin', false);
$$;

-- An examiner reads/writes ONLY the rows they own (`owner_id = auth.uid()`).
-- A supervisor (app_metadata.role = 'admin') reads and manages every row.
-- There is no `anon` policy at all: an unauthenticated client is rejected by
-- the database, so the publishable key can never pull the children's dataset.
create policy "albasar_select_own_or_admin"
  on public.student_screening_records
  for select
  to authenticated
  using (owner_id = auth.uid() or public.is_albasar_admin());

create policy "albasar_insert_own_or_admin"
  on public.student_screening_records
  for insert
  to authenticated
  with check (owner_id = auth.uid() or public.is_albasar_admin());

create policy "albasar_update_own_or_admin"
  on public.student_screening_records
  for update
  to authenticated
  using (owner_id = auth.uid() or public.is_albasar_admin())
  with check (owner_id = auth.uid() or public.is_albasar_admin());

-- There is intentionally NO delete policy: RLS denies by default, so a
-- synced research record can never be removed — not even by a supervisor.
-- ALCOA+ / SPIRIT-AI audit trail is preserved.

-- Table privileges are separate from RLS policies. Without these GRANTs the
-- Data API returns "permission denied for table" even when policies exist.
-- Only `authenticated` is granted here; `anon` is intentionally left out.
grant select, insert, update on public.student_screening_records to authenticated;
revoke all on public.student_screening_records from anon;

-- =========================================================
-- Ocular photo storage (PRIVATE bucket — IRB compliant)
-- =========================================================
-- The examiner app compresses each ocular photo to 1024 px / q0.85 and stores
-- it offline in IndexedDB. During cloud sync the blob is uploaded under the
-- folder `{authUserId}/{childId}/{timestamp}.jpg`, and only that OBJECT PATH
-- is written to `eye_photo_url`. The app resolves the path into a signed URL
-- (1 hour) on demand through `createSignedUrl()`.
insert into storage.buckets (id, name, public)
values ('screening-photos', 'screening-photos', false)
on conflict (id) do update set public = false;

drop policy if exists "albasar photo open read" on storage.objects;
drop policy if exists "albasar photo open insert" on storage.objects;
drop policy if exists "albasar photo open update" on storage.objects;
drop policy if exists "albasar_photo_select_own_or_admin" on storage.objects;
drop policy if exists "albasar_photo_insert_own_or_admin" on storage.objects;
drop policy if exists "albasar_photo_update_own_or_admin" on storage.objects;

create policy "albasar_photo_select_own_or_admin"
  on storage.objects
  for select
  to authenticated
  using (
    bucket_id = 'screening-photos'
    and ((storage.foldername(name))[1] = auth.uid()::text or public.is_albasar_admin())
  );

create policy "albasar_photo_insert_own_or_admin"
  on storage.objects
  for insert
  to authenticated
  with check (
    bucket_id = 'screening-photos'
    and ((storage.foldername(name))[1] = auth.uid()::text or public.is_albasar_admin())
  );

create policy "albasar_photo_update_own_or_admin"
  on storage.objects
  for update
  to authenticated
  using (
    bucket_id = 'screening-photos'
    and ((storage.foldername(name))[1] = auth.uid()::text or public.is_albasar_admin())
  )
  with check (
    bucket_id = 'screening-photos'
    and ((storage.foldername(name))[1] = auth.uid()::text or public.is_albasar_admin())
  );

-- ⚠️ لا تُنفّذ "grant all on storage.objects to anon, authenticated" هنا:
--    - على Supabase Cloud، storage.objects مملوك لحساب postgres (storage.super_user)
--      وقد منحه Supabase أصلاً كل الصلاحيات لـ anon/authenticated
--      (migration 0046)؛ فالسطر كان تكراراً بلا فائدة.
--    - grant all يمنح DELETE غير مبرّر، ويجعل السكريبت يفشل بـ 42501
--      "must be owner of table objects" في أي مشروع غيّر فيه storage.super_user.
--
-- 📋 أخلاقيات البحث (IRB): الحاوية خاصة (public = false). صور عيون الأطفال
--    بيانات بيومترية حساسة؛ لا تُقرأ إلا برابط موقّع صالح ساعة واحدة، ولا
--    يرفعها الفاحص إلا داخل مجلده {auth.uid()}/…، ولا يطّلع عليها المشرف
--    إلا عبر نفس الواجهة.

-- لا توجد سياسة DELETE على storage.objects إطلاقاً.

-- صلاحيات جدول storage (اختيارية + لا تُفشل السكريبت): Supabase منح
-- anon/authenticated صلاحيات SELECT/INSERT/UPDATE/DELETE أصلاً (migration 0046).
-- هذا المقطع يضمن الـ SELECT/INSERT/UPDATE فقط إن اختلفت إعدادات المشروع،
-- ولا يقوّي DELETE أبداً. حماية القراءة الفعلية تأتي من سياسات RLS أعلاه
-- (لا توجد سياسة لـ anon).
do $$
begin
  grant select, insert, update on storage.objects to anon, authenticated;
exception
  when insufficient_privilege then
    raise notice 'تعذّر منح صلاحيات storage.objects (غير مالك للجدول) — سياسات RLS أعلاه تكفي.';
end $$;

-- Al-Basar AI · الأعمدة الناقصة + حاوية صور العين (خاصة)
-- (بدون المساس بالبيانات الحالية — آمن للتشغيل المتكرر)
--
-- ⚠️ اسم الجدول الصحيح هو public.student_screening_records
--    (وليس public.screening_records — وهو غير موجود وسيُتخطى بصمت)
--
-- ✅ لا يحتوي هذا الملف على "grant all on storage.objects to anon, authenticated":
--    - على Supabase Cloud، storage.objects مملوك لحساب postgres (storage.super_user)
--      وقد منحه Supabase أصلاً كل الصلاحيات لـ anon/authenticated
--      (migration 0046)؛ فالسطر كان تكراراً بلا فائدة.
--    - grant all يمنح DELETE غير مبرّر، ويجعل السكريبت يفشل بـ 42501
--      "must be owner of table objects" في أي مشروع غيّر فيه storage.super_user.
--    - الوصول للصور يُدار بالكامل عبر سياسات RLS أدناه.
--
-- ✅ سياسات جدول السجلات الآن مقيّدة بـ owner_id = auth.uid():
--    كل فاحص يرى سجلاته فقط، والمشرف (app_metadata.role = 'admin') يرى الكل.
--    لا توجد سياسة DELETE إطلاقاً.
--
-- ✅ حاوية الصور private (public = false) — أخلاقيات البحث (IRB):
--    صور عيون الأطفال بيانات بيومترية حساسة؛ لا تُقرأ إلا عبر Signed URL
--    صالحة ساعة واحدة، ويصعدها الفاحص داخل مجلده {auth.uid()}/… فقط.
--
-- ✅ سياسة UPDATE تحتوي USING و WITH CHECK معاً (شرط PostgreSQL)،
--    فيُسمح بإعادة تصوير نفس التلميذ عبر upsert.
-- ============================================================

-- ---------- 1) دالة الصلاحية (تقرأ app_metadata حصراً) ----------
-- app_metadata محفوظ في auth.users.raw_app_meta_data ولا يعدّله المستخدم
-- (updateUser يعدّل user_metadata فقط). لذلك هي المصدر الوحيد الموثوق.
create or replace function public.is_albasar_admin()
returns boolean
language sql
stable
as $$
  select coalesce((auth.jwt() -> 'app_metadata' ->> 'role') = 'admin', false);
$$;

-- ---------- 2 + 3) الأعمدة الناقصة + RLS الصارم على جدول السجلات ----------
-- كل ما يخص جدول السجلات داخل كتلة DO واحدة: إن كان الجدول غير موجود
-- (مشروع جديد) تُتخطى هذه الخطوات برسالة واضحة بدل أن يفشل السكربت
-- بالكامل ويلغي إنشاء الحاوية والسياسات (SQL Editor يعمل داخل معاملة).
do $$
begin
  if to_regclass('public.student_screening_records') is null then
    raise notice 'الجدول public.student_screening_records غير موجود — تُتخطى خطوات الجدول. شغّل schema.sql أولاً (Bootstrap كامل).';
    return;
  end if;

  alter table public.student_screening_records
    add column if not exists external_eye_findings jsonb default '[]'::jsonb,
    add column if not exists eye_photo_url text,
    add column if not exists owner_id uuid default auth.uid(),
    add column if not exists examiner_full_name text,
    add column if not exists device_id text,
    add column if not exists examination_date timestamptz;

  alter table public.student_screening_records enable row level security;

  -- 🔧 تصحيح نوع عمودَي الزمن (SQLSTATE 22P02):
  --    المؤقتان يسجّلان الزمن بعُشر الثانية (74.5 مثلاً)، والعمود كان integer
  --    فيرفضه PostgreSQL بـ «invalid input syntax for type integer: "74.5"».
  --    `add column if not exists` لا يغيّر نوع عمود موجود، لذا التحويل صريح.
  alter table public.student_screening_records
    alter column paper_time_sec type numeric using paper_time_sec::numeric,
    alter column smart_time_sec type numeric using smart_time_sec::numeric;

  -- إسقاط أي سياسة مفتوحة سابقة (تمنع تسريب بيانات التلاميذ عبر anon).
  drop policy if exists "albasar open select" on public.student_screening_records;
  drop policy if exists "albasar open insert" on public.student_screening_records;
  drop policy if exists "albasar open update" on public.student_screening_records;
  drop policy if exists "allow_anon_all" on public.student_screening_records;
  -- أسماء anon قديمة وُجدت فعلاً على الخادم الحي (إصدارات أقدم من المخطط).
  drop policy if exists "albasar anon select" on public.student_screening_records;
  drop policy if exists "albasar anon insert" on public.student_screening_records;
  drop policy if exists "albasar anon update" on public.student_screening_records;
  drop policy if exists "albasar anon delete" on public.student_screening_records;
  drop policy if exists "albasar_select_own_or_admin" on public.student_screening_records;
  drop policy if exists "albasar_insert_own_or_admin" on public.student_screening_records;
  drop policy if exists "albasar_update_own_or_admin" on public.student_screening_records;

  -- الفاحص يرى/يضيف/يحدث سجلاته فقط، والمشرف يرى ويدير كل سجلات الطلبة.
  create policy "albasar_select_own_or_admin"
    on public.student_screening_records for select
    to authenticated
    using (owner_id = auth.uid() or public.is_albasar_admin());

  create policy "albasar_insert_own_or_admin"

/**
 * Returns a valid JWT for the signed-in examiner, or `null` when there is no
 * session.
 *
 * ⚠️ The public anon key is deliberately **never** returned here. The cloud
 * dataset is protected by `owner_id = auth.uid()` RLS policies, so an
 * anonymous write would either be rejected by the database or — worse —
 * require the very table-wide SELECT/UPDATE grants that expose every
 * child's record. Offline work stays in the device's local store until the
 * examiner signs in.
 */
export async function getAuthAccessToken(): Promise<string | null> {
  const client = getSupabaseClient();
  if (!client) return null;

  try {
    const { data } = await client.auth.getSession();
    const session = data.session;
    if (!session?.access_token) return null;

    const expiresAt = session.expires_at ? session.expires_at * 1000 : 0;
    if (expiresAt && expiresAt - Date.now() < 5 * 60 * 1000) {
      try {
        const { data: refreshed, error } = await client.auth.refreshSession();
        if (!error && refreshed.session?.access_token) {
          return refreshed.session.access_token;
        }
      } catch {
        // fall through to the (possibly expired) cached token
      }
    }

    return session.access_token;
  } catch {
    return null;
  }
}

/** Convert undefined to null so PostgREST receives explicit SQL NULL values. */
const n = <T>(value: T | undefined): T | null => (value === undefined ? null : value);

/**
 * Coerce a value to a finite number, or `null`.
 *
 * PostgreSQL rejects the string `"74.5"` for an `integer` column with
 * SQLSTATE `22P02` (`invalid input syntax for type integer`). Both stopwatches
 * report one decimal (`elapsed.toFixed(1)`), so every numeric column is
 * funnelled through one of the helpers below and a stray decimal or a
 * non-numeric string can never reach the database as a type error.
 *
 * Empty strings and `null`/`undefined` map to SQL `NULL` rather than `0`,
 * because `Number(null)` and `Number('')` are both `0` and would silently
 * fabricate a measurement.
 */
const toSafeNum = (value: unknown): number | null => {
  if (value === null || value === undefined || value === '') return null;
  const parsed = typeof value === 'number' ? value : Number(value);
  return Number.isFinite(parsed) ? parsed : null;
};

/**
 * Coerce a value to a finite integer, or `null`.
 *
 * Used for every `integer` column (`grade_level`, `parental_consent`,
 * `wears_glasses`, `paper_referral`, `hirschberg_prism_diopters`,
 * `triage_ocular_flag`, `ai_quality_flag`, `asthenopia_score`,
 * `final_triage_referral`, `follow_up_status`, …).
 *
 * `Math.round` keeps 0/1 flags and small counts exact while guaranteeing the
 * value satisfies the column's integer type.
 */
const toSafeInt = (value: unknown): number | null => {
  const parsed = toSafeNum(value);
  return parsed === null ? null : Math.round(parsed);
};

/**
 * Coerce a value to a number that fits PostgreSQL `numeric(6,3)`, or `null`.
 *
 * `numeric(6,3)` accepts at most `999.999`; a larger value fails with
 * SQLSTATE `22003` (`numeric field overflow`). Clamping keeps the sync alive
 * instead of aborting the whole record.
 */
const toSafeNum63 = (value: unknown): number | null => {
  const parsed = toSafeNum(value);
  if (parsed === null) return null;
  return Math.min(999.999, Math.max(-999.999, parsed));
};

/**
 * Fetch every screening row the current role may read. Uses the Supabase
 * client so PostgreSQL RLS decides visibility:
 *   * an examiner sees only the rows whose `owner_id` matches their `auth.uid()`
 *   * an admin (`app_metadata.role = 'admin'`) sees every row
 *
 * There is no anon fallback: an unauthenticated call is rejected by the
 * database rather than silently widening the blast radius.
 */
export async function fetchScreeningRecords(): Promise<StudentScreeningRecord[]> {
  const client = getSupabaseClient();
  if (!client) throw new Error('Supabase client is not available.');

  const token = await getAuthAccessToken();
  if (!token) {
    throw new Error('تسجيل الدخول مطلوب لقراءة السجلات السحابية.');
  }

  const { data, error } = await client.from(TABLE).select('*');
  if (error) {
    throw new Error(`HTTP ${(error as { status?: number }).status ?? ''}: ${error.message}`);
  }
  const rows = (data ?? []) as Record<string, unknown>[];
  return rows.map(mapSupabaseRowToRecord);
}

/**
 * Resolve a stored eye-photo path into a temporary, signed URL.
 *
 * The `screening-photos` bucket is **private** (IRB): the object path is the
 * only thing persisted in `eye_photo_url`, and every view resolves it through
 * a short-lived signed URL. `path` may also be a legacy public URL, in which
 * case it is returned unchanged.
 */
export async function createSignedPhotoUrl(
  path: string | undefined | null,
  expiresInSeconds = 3600,
): Promise<string | null> {
  const value = (path ?? '').trim();
  if (!value) return null;
  if (/^https?:\/\//i.test(value)) return value; // legacy public URL

  const client = getSupabaseClient();
  if (!client) return null;
  const token = await getAuthAccessToken();
  if (!token) return null;

  try {
    const { data, error } = await client.storage
      .from('screening-photos')
      .createSignedUrl(value, expiresInSeconds);
    if (error) return null;
    return data?.signedUrl ?? null;
  } catch {
    return null;
  }
}

function extractMissingColumn(message: string): string | null {
  const patterns = [
    /Could not find the '([^']+)' column/i,
    /column\s+[a-zA-Z0-9_]+\.([a-zA-Z0-9_]+)\s+does not exist/i,
    /column\s+"([^"]+)"\s+does not exist/i,

function formatError(status: number, body: PgErrorBody): string {

  const details = body.details ? ` — ${body.details}` : '';

 * Upsert one screening record through the PostgREST endpoint. The function
 * surfaces the exact Supabase error text and, when PostgREST reports a stale
 * schema cache for an optional column, removes that column and retries instead
 * of silently swallowing the failure.
 *
 * ⚠️ A real, signed-in session is **required**. The previous implementation
 * silently downgraded to the public anon key on a 401/403, which combined
 * with table-wide SELECT/UPDATE grants made the whole children's dataset
 * readable by anyone holding the publishable key. Records created offline stay
 * queued in the device's local store until the examiner signs in.
 *
 * @param ownerId The signed-in Supabase account id written to `owner_id`.
 */
export async function upsertScreeningRecord(
  record: StudentScreeningRecord,
  ownerId?: string | null,
): Promise<SupabaseDiagnostic> {
  if (!isSupabaseConfigured()) {
    throw new Error('Supabase is not configured: missing project URL or anon key.');
  }

  const token = await getAuthAccessToken();
  if (!token) {
    throw new Error('يلزم تسجيل الدخول قبل مزامنة السجلات — السجلات محفوظة محلياً بأمان.');
  }

  const payload = mapRecordToSupabaseRow(record, ownerId);
  const droppedColumns: string[] = [];
  const url = `${SUPABASE_URL_VALUE}/rest/v1/${TABLE}?on_conflict=child_id`;

  for (let attempt = 0; attempt < 25; attempt += 1) {
    let response: Response;
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 30000);
    try {
      response = await fetch(url, {
        method: 'POST',
        signal: controller.signal,
        headers: {
          apikey: SUPABASE_ANON_KEY_VALUE,
          Authorization: `Bearer ${token}`,
          'Content-Type': 'application/json',
          Prefer: 'resolution=merge-duplicates,return=minimal',
        },
        body: JSON.stringify(payload),
      });
    } catch (networkError) {
      const aborted = networkError instanceof DOMException && networkError.name === 'AbortError';
      throw new Error(
        aborted
          ? 'Network error while contacting Supabase: request timed out after 30 seconds.'
          : `Network error while contacting Supabase: ${networkError instanceof Error ? networkError.message : String(networkError)}`,
      );
    } finally {
      clearTimeout(timeout);
    }

    if (response.ok) {
      return { ok: true, status: response.status, message: 'Upsert succeeded.', droppedColumns };
    }

    const text = await response.text();
    const body = parsePgError(text);

    // A stale or revoked JWT is a hard stop: it is never retried with the
    // public anon key, because that would defeat the `owner_id` policies.
    if (response.status === 401 || response.status === 403) {
      const error = new Error(
        `HTTP ${response.status}${body.code ? ` (${body.code})` : ''}: ${
          response.status === 401
            ? 'انتهت صلاحية الجلسة. سجّل الدخول مجدداً ثم أعد المزامنة — السجلات محفوظة محلياً.'
            : 'رفضت الخادم عملية الكتابة. تحقّق من سياسات RLS أو من صلاحية حسابك.'
        }`,
      );
      (error as Error & { status?: number; code?: string }).status = response.status;
      (error as Error & { status?: number; code?: string }).code = body.code;
      throw error;
    }

    const missingColumn = extractMissingColumn(body.message ?? '');
    if (missingColumn && Object.prototype.hasOwnProperty.call(payload, missingColumn)) {
      delete payload[missingColumn];
      droppedColumns.push(missingColumn);
      continue;
    }

    const error = new Error(formatError(response.status, body));
    (error as Error & { status?: number; code?: string }).status = response.status;
    (error as Error & { status?: number; code?: string }).code = body.code;
    throw error;
  }

  throw new Error(
    `HTTP 400: Too many schema mismatches while syncing. Dropped columns: ${droppedColumns.join(', ') || 'none'}`,
  );
}

/** Encodes an ArrayBuffer as an unpadded Base64URL string. */
export const bufferToBase64Url = (buffer: ArrayBuffer): string => {
  const bytes = new Uint8Array(buffer);
  let binary = '';
  for (let i = 0; i < bytes.length; i += 1) binary += String.fromCharCode(bytes[i]);
  return btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
};

/** Decodes a Base64URL string back into an ArrayBuffer. */
export const base64UrlToBuffer = (value: string): ArrayBuffer => {
  const normalized = value.replace(/-/g, '+').replace(/_/g, '/');
  const padded = normalized + '='.repeat((4 - (normalized.length % 4)) % 4);
  const binary = atob(padded);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i += 1) bytes[i] = binary.charCodeAt(i);
  return bytes.buffer;
};

/** A fresh 32-byte challenge. Backed by a plain ArrayBuffer so it satisfies `BufferSource`. */
const randomChallenge = (): Uint8Array<ArrayBuffer> => {
  const buffer = new ArrayBuffer(32);
  const bytes = new Uint8Array(buffer);
  if (typeof crypto !== 'undefined' && typeof crypto.getRandomValues === 'function') {
    crypto.getRandomValues(bytes);
  } else {
    for (let i = 0; i < bytes.length; i += 1) bytes[i] = Math.floor(Math.random() * 256);
  }
  return bytes;
};

/**
 * Creates a platform-bound, resident credential for the signed-in examiner and
 * stores only its public credential id.
 *
 * Returns `true` when the OS confirmed the biometric and the binding was
 * persisted. Returns `false` when the device has no sensor, the user
 * cancelled, or the platform refused — it never throws.
 */
export async function registerBiometricCredential(
  userId: string,
  displayName: string,
): Promise<boolean> {
  if (!isSecureAuthContext()) return false;
  if (typeof navigator === 'undefined' || !navigator.credentials) return false;
  if (typeof PublicKeyCredential === 'undefined') return false;

  const safeUserId = userId.trim() || 'albasar-examiner';
  const safeDisplayName = displayName.trim() || safeUserId;

  try {
    const userIdBytes = new TextEncoder().encode(safeUserId);

    const publicKey: PublicKeyCredentialCreationOptions = {
      challenge: randomChallenge(),
      rp: { name: 'Al-Basar AI', id: window.location.hostname },
      user: {
        id: userIdBytes,
        name: safeUserId,
        displayName: safeDisplayName,
      },
      pubKeyCredParams: [
        { type: 'public-key', alg: -7 },   // ES256
        { type: 'public-key', alg: -257 }, // RS256
      ],
      authenticatorSelection: {
        authenticatorAttachment: 'platform',
        residentKey: 'required',
        requireResidentKey: true,
        userVerification: 'required',
      },
      timeout: BIOMETRIC_TIMEOUT_MS,
      attestation: 'none',
    };

    const credential = await navigator.credentials.create({ publicKey });
    if (!credential) return false;

    const publicCredential = credential as PublicKeyCredential;
    // `rawId` is the canonical binary credential id; encoding it ourselves
    // guarantees the value we store decodes back to the exact same bytes.
    const credentialId = publicCredential.rawId
      ? bufferToBase64Url(publicCredential.rawId)
      : publicCredential.id;

    if (!credentialId) return false;

    const binding: BiometricBinding = {
      credentialId,
      userId: safeUserId,
      displayName: safeDisplayName,
      createdAt: new Date().toISOString(),
    };
    safeSet(BIOMETRIC_CREDENTIAL_KEY, JSON.stringify(binding));
    return true;
  } catch {
    // Cancelled, no sensor, or the authenticator refused — fall back silently.
    return false;
  }
}

  if (typeof navigator === 'undefined' || !navigator.credentials) return false;
  if (typeof PublicKeyCredential === 'undefined') return false;

  const buildRequest = (allowCredentials: PublicKeyCredentialDescriptor[]): PublicKeyCredentialRequestOptions => ({
    challenge: randomChallenge(),
    rpId: window.location.hostname,
    allowCredentials,
    userVerification: 'required',
    timeout: BIOMETRIC_TIMEOUT_MS,
  });

  const scopedDescriptors: PublicKeyCredentialDescriptor[] = [{
    id: base64UrlToBuffer(binding.credentialId),
    type: 'public-key',
  }];

  try {
    const assertion = await navigator.credentials.get({ publicKey: buildRequest(scopedDescriptors) });
    return Boolean(assertion);
  } catch (error) {
    // The user dismissed the OS sheet — never re-prompt behind their back.
    if (isUserCancellation(error)) return false;
    // Some platforms expose the credential as purely discoverable; retry once
    // without the explicit allowCredentials list before giving up.
    try {
      const assertion = await navigator.credentials.get({ publicKey: buildRequest([]) });
      return Boolean(assertion);
    } catch {
      return false;
    }
  }
}

  isOnline: boolean;
  isSyncing: boolean;
  cloudUser: CloudAuthUser | null;
  onOpenLogin: () => void;
  onSync: () => void;
  onPullLatest: () => void;
}

/**
 * Supervisor-only technical surface. The examiner-facing modal stays free of
 * Supabase plumbing; every configuration detail lives here, inside the
 * role-gated admin drawer.
 */
export const CloudSettingsPanel: React.FC<CloudSettingsPanelProps> = ({
  isOnline,
  isSyncing,
  cloudUser,
  onOpenLogin,
  onSync,
  onPullLatest,

  const [anonKey, setAnonKey] = useState(SUPABASE_ANON_KEY_VALUE);
  const [testing, setTesting] = useState(false);
  const [testResult, setTestResult] = useState<string | null>(null);
  const [savedHint, setSavedHint] = useState<string | null>(null);

    setTestResult(null);

      const diag = await testSupabaseConnection(url, anonKey);

          : `❌ HTTP ${diag.status ?? ''}${diag.code ? ` (${diag.code})` : ''}: ${diag.message}`,
      );

      setTestResult(`❌ خطأ: ${error instanceof Error ? error.message : String(error)}`);
    } finally {
      setTesting(false);
    }

  return (
    <div className="cloud-settings">
      <div className="cloud-settings__status">

        </span>
      </div>

      {!isOnline && (
        <div className="notice notice--warn" role="status">
          <WifiOff size={16} aria-hidden="true" />

          تسجيل الدخول
        </button>
        <button type="button" onClick={onSync} className="m3-btn" disabled={!isOnline || isSyncing}>
          {isSyncing ? 'جاري المزامنة…' : 'مزامنة الآن'}
        </button>
        <button type="button" onClick={onPullLatest} className="m3-btn" disabled={!isOnline || isSyncing}>
          <CloudDownload size={16} aria-hidden="true" />
          سحب أحدث السجلات من السحابة
        </button>
      </div>

      <details className="cloud-settings__details">
        <summary>إعدادات Supabase (للمشرف فقط)</summary>

        <label className="auth-field">
          <span>Supabase URL</span>
          <span className="auth-input">
            <input dir="ltr" value={url} onChange={(e) => setUrl(e.target.value)} placeholder="https://your-project.supabase.co" />
          </span>
        </label>

        <label className="auth-field">
          <span>Anon Key</span>
          <span className="auth-input">
            <input dir="ltr" value={anonKey} onChange={(e) => setAnonKey(e.target.value)} placeholder="sb_publishable_..." />
          </span>
        </label>

        <div className="cloud-settings__actions">
          <button type="button" className="m3-btn" onClick={handleSave}>
            <Save size={16} aria-hidden="true" />
            حفظ الإعدادات
          </button>
          <button type="button" className="m3-btn" onClick={handleTest} disabled={testing || !isOnline}>
            <RefreshCw size={16} className={testing ? 'spin' : ''} aria-hidden="true" />
            {testing ? 'جاري الفحص…' : 'فحص الاتصال'}
          </button>
        </div>

        {savedHint && <p className="cloud-settings__hint">{savedHint}</p>}
        {testResult && <p className="cloud-settings__hint">{testResult}</p>}
      </details>
    </div>
  );
};
